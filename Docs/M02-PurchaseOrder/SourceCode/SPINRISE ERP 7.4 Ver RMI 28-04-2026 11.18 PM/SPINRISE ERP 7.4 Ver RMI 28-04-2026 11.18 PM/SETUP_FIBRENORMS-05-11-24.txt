	Ksp_SetupFibrenorms '01'
*/
ALTER PROCEDURE KSP_SETUPFIBRENORMS  
(
@DIVCODE VARCHAR(02)   
)
AS    
BEGIN    
    
SELECT a.DIVCODE,VARNAME ,SPANLEN25,SPANLEN50,UR,MIC,FIBRESTENGTH,ELONGATION,TRASHPER,    
SCI,SFI,IFC,NEPS,SCN,a.ML,a.UHML,a.UI,a.GTEX,maturityratio,FQI,SFC,pd.DIV_PRINTNAME,pd.DIV_UNITNAME 
FROM RM_FIBRENORMS A    
INNER JOIN RM_VAR C on a.VARCODE=c.VARCODE    
INNER JOIN PP_DIVMAS pd on pd.DIVCODE=a.DivCode    
WHERE a.DivCode=@divcode     
ORDER BY a.VARCODE    

end
