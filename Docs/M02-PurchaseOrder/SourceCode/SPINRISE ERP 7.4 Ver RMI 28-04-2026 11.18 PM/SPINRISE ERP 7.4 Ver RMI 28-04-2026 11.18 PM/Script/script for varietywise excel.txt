--EXEC CRY_STKSTMNTTYPEWISE '01','2016-06-30','2016-06-30','2016-04-01','2017-03-31' ,'2016','A','T'                            
--EXEC CRY_datevarietywise_excel '02','2019-12-01','2019-12-30','2019-04-01','2020-03-31' ,'2019','A','z'                            
        
 --EXEC CRY_varietywise_excel '02','2019-12-01','2019-12-30','2019-04-01','2020-03-31' ,'2019'                          
      
        
        
CREATE PROC CRY_varietywise_excel                           
@DIVCODE VARCHAR(5),                            
@FROMDATE VARCHAR(15),                            
@TODATE VARCHAR(15),                            
@YFDATE VARCHAR(15),                            
@YLDATE VARCHAR(15),                            
@TYEAR VARCHAR(5),                           
@FROMCAT varchar(5),                            
@TOCAT varchar(5)                            
AS                             
                            
                            
select D.CATNAME,C.VARNAME,                             
CASE WHEN sum(isnull(a.OPENBAL,0))>0 THEN sum(isnull(a.OPENBAL,0)) ELSE 0 END as opbales,                            
CASE WHEN sum(isnull(a.OPENBOR,0)) >0 THEN sum(isnull(a.OPENBOR,0)) ELSE 0 END as opbORAS,                            
CASE WHEN sum(isnull(a.OPENKGS,0)) >0 THEN sum(isnull(a.OPENKGS,0)) ELSE 0 END as OPKGS,                            
CASE WHEN sum(isnull(a.OPENVALUE,0))>0 THEN sum(isnull(a.OPENVALUE,0)) ELSE 0 END as OPVALUE,                            
-- receipt        
SUM(ISNULL(A.RECBAL,0))+SUM(ISNULL(A.ISSRETBAL,0)) AS RECBALES,        
SUM(ISNULL(A.RECBOR,0))+SUM(ISNULL(A.ISSRETBOR,0)) AS RECBORAS,                            
SUM(ISNULL(A.RECKGS,0))+SUM(ISNULL(A.ISSRETKGS,0)) AS RECKGS,        
SUM(ISNULL(A.RECVALUE,0))+SUM(ISNULL(A.ISSRETVAL,0))  AS RECVALUE,          
-- other receipt        
         
 (SUM(ISNULL(A.JOBRECBAL,0))+SUM(ISNULL(A.TRRECBAL,0))+SUM(ISNULL(A.ISSRETBAL,0))  ) AS OTHERRECBALES,                                             
 (SUM(ISNULL(A.JOBRECKGS,0)) +SUM(ISNULL(A.TRRECKGS,0)) +SUM(ISNULL(A.ISSRETKGS,0))  ) AS OTHERRECKGS,                                             
 (SUM(ISNULL(A.JOBRECVALUE,0)) + SUM(ISNULL(A.TRRECVALUE,0)) + SUM(ISNULL(A.ISSRETVAL,0))   ) AS OTHERRECVALUE,                                             
        
SUM(ISNULL(A.JOBRECBAL,0)) AS JOBRECBALES,        
SUM(ISNULL(A.JOBRECBOR,0)) AS JOBRECBOR,                            
SUM(ISNULL(A.JOBRECKGS,0)) AS JOBRECKGS,        
SUM(ISNULL(A.JOBRECVALUE,0)) AS JOBRECVALUE,                            
                            
SUM(ISNULL(A.TRRECBAL,0)) AS TRRECBALES,        
SUM(ISNULL(A.TRRECBOR,0)) AS TRRECBORAS,                            
SUM(ISNULL(A.TRRECKGS,0)) AS TRRECKGS,        
SUM(ISNULL(A.TRRECVALUE,0)) AS TRRECVALUE,                           
                         
SUM(ISNULL(A.ISSRETBAL,0)) AS ISSRETBAL,        
SUM(ISNULL(A.ISSRETBOR,0)) AS ISSRETBOR,                            
SUM(ISNULL(A.ISSRETKGS,0)) AS ISSRETKGS,        
SUM(ISNULL(A.ISSRETVAL,0)) AS ISSRETVAL,         
        
        
        
  -- issue                             
SUM(ISNULL(A.PRODBAL,0))  AS ISSBALES,                            
SUM(ISNULL(A.PRODBOR,0)) AS ISSBORAS,                            
SUM(ISNULL(A.PRODKGS,0)) AS ISSKGS,                            
SUM(ISNULL(A.PRODVALUE,0)) AS ISSVALUE,            
-- other issue         
(SUM(ISNULL(A.SALESBAL,0)) + SUM(ISNULL(A.JWBAL,0)) +  SUM(ISNULL(A.TRANSBAL,0)) +SUM(ISNULL(A.REJBAL,0))+ SUM(ISNULL( ISSKNITBAL ,0))+SUM(isnull(ISSRETKNITBAL,0)) ) AS OTHERISSUEBALES,        
 ( SUM(ISNULL(A.SALESKGS,0)) + SUM(ISNULL(A.JWKGS,0)) +  SUM(ISNULL(A.TRANSKGS,0))  + SUM(ISNULL(A.REJKGS,0)) +  sum(ISNULL(ISSKNITKGS,0)) +   sum(ISNULL(ISSRETKNITKGS,0))  ) AS OTHERISSUEKGS,        
(SUM(ISNULL(A.SALESVALUE,0)) + SUM(ISNULL(A.JWVALUE,0))  +  SUM(ISNULL(A.TRANSVALUE,0)) +SUM(ISNULL(A.REJVALUE,0))+ SUM(ISNULL(ISSKNITVAL,0)) +  SUM(ISNULL(ISSRETKINTVAL,0)) ) AS OTHERISSUEVALUE,        
                      
SUM(ISNULL(A.SALESBAL,0)) AS SALESBAL,        
SUM(ISNULL(A.SALESBOR,0)) AS SALESBOR,        
SUM(ISNULL(A.SALESKGS,0))AS SALESKGS,                            
SUM(ISNULL(A.SALESVALUE,0)) AS SALESVALUE ,          
                          
SUM(ISNULL(A.JWBAL,0)) AS JWISSBAL,                            
SUM(ISNULL(A.JWBOR,0)) AS JWISSBOR,        
SUM(ISNULL(A.JWKGS,0)) AS JWISSKGS,        
SUM(ISNULL(A.JWVALUE,0)) AS JWISSVAL,         
                           
SUM(ISNULL(A.TRANSBAL,0)) AS TRANSBAL,        
SUM(ISNULL(A.TRANSBOR,0)) AS TRANSBOR,                            
SUM(ISNULL(A.TRANSKGS,0)) as TRANSKGS,        
SUM(ISNULL(A.TRANSVALUE,0)) AS TRANSVALUE,         
                           
SUM(ISNULL(A.REJBAL,0)) AS REJBALES,        
SUM(ISNULL(A.REJKGS,0)) AS REJKGS,                            
SUM(ISNULL(A.REJVALUE,0)) AS REJVALUE,         
               
 SUM(ISNULL( ISSKNITBAL ,0))AS ISSKNITBAL,        
 SUM(ISNULL(ISSKNITBOR ,0)) AS ISSKNITBOR,        
 sum(ISNULL(ISSKNITKGS,0)) AS ISSKNITKGS,              
 SUM(ISNULL(ISSKNITVAL,0)) AS ISSKNITVAL          
                     
 ,SUM(isnull(ISSRETKNITBAL,0)) ISSRETKNITBAL ,        
 SUM(ISNULL(ISSRETKNITBOR,0)) ISSRETKNITBOR,        
 sum(ISNULL(ISSRETKNITKGS,0)) ISSRETKNITKGS,              
 SUM(ISNULL(ISSRETKINTVAL,0)) ISSRETKINTVAL,                 
               
    SUM(ISNULL(A.CTRECBAL,0)) AS CTRECBAL,        
SUM(ISNULL(A.CTRECBOR,0)) AS CTRECBOR,                                                
SUM(ISNULL(A.CTRECKGS,0)) AS CTRECKGS,        
SUM(ISNULL(A.CTRECVALUE,0)) AS CTRECVALUE,           
                            
--case when sum(isnull(a.OPENBAL,0)) > 0 then                    
-- ((sum(isnull(a.OPENBAL,0))+sum(isnull(a.recBAL,0))+SUM(ISNULL(A.JOBRECBAL,0))+                        
-- SUM(ISNULL(A.TRRECBAL,0))+ SUM(ISNULL(A.ISSRETBAL,0)))-                            
-- (sum(isnull(a.PRODBAL,0))+ sum(isnull(a.TRANSBAL,0))+sum(isnull(a.SALESBAL,0))+                            
-- sum(isnull(a.JWBAL,0))+sum(isnull(a.REJBAL,0))))           
-- else                             
-- ((sum(isnull(a.recBAL,0))+ SUM(ISNULL(A.TRRECBAL,0)) + SUM(ISNULL(A.JOBRECBAL,0))+ SUM(ISNULL(A.ISSRETBAL,0)) ) -                            
-- (sum(isnull(a.PRODBAL,0))+ sum(isnull(a.TRANSBAL,0))+sum(isnull(a.SALESBAL,0))+sum(isnull(a.JWBAL,0))                             
-- +sum(isnull(a.REJBAL,0)))) END           
           
-- closing        
        
 CASE WHEN sum(isnull(a.OPENBAL,0))>0 THEN sum(isnull(a.OPENBAL,0)) ELSE 0 END+SUM(ISNULL(A.RECBAL,0))+SUM(ISNULL(A.ISSRETBAL,0))+ SUM(isnull(ISSRETKNITBAL,0))                 
 +SUM(ISNULL(A.JOBRECBAL,0))+SUM(ISNULL(A.TRRECBAL,0))-(SUM(ISNULL(A.PRODBAL,0))+ SUM(ISNULL( ISSKNITBAL ,0))+SUM(ISNULL(A.JWBAL,0))+SUM(ISNULL(A.TRANSBAL,0))+                  
 SUM(ISNULL(A.SALESBAL,0)))                  
                   
  CLOBALES,                            
                            
case when sum(isnull(a.OPENBOR,0)) > 0 then                             
    ((sum(isnull(a.OPENBOR,0))+sum(isnull(a.recBOR,0))+SUM(ISNULL(A.JOBRECBOR,0))+ SUM(ISNULL(A.TRRECBOR,0))+SUM(ISNULL(A.ISSRETBOR,0))+SUM(ISNULL(ISSRETKNITBOR,0)))-                            
          (sum(isnull(a.PRODBOR,0))+SUM(ISNULL(ISSKNITBOR ,0))+ sum(isnull(a.TRANSBOR,0))+sum(isnull(a.SALESBOR,0))+sum(isnull(a.JWBOR,0))))                            
       else ((sum(isnull(a.recBOR,0))+ SUM(ISNULL(A.JOBRECBOR,0))+ SUM(ISNULL(A.TRRECBOR,0))+SUM(ISNULL(A.ISSRETBOR,0))+SUM(ISNULL(ISSRETKNITBOR,0)))-                            
       (sum(isnull(a.PRODBOR,0))+SUM(ISNULL(ISSKNITBOR ,0))+ sum(isnull(a.TRANSBOR,0))+sum(isnull(a.SALESBOR,0))+sum(isnull(a.JWBOR,0))))                            
        END CLOBORAS,                            
                                    
case when sum(isnull(a.OPENKGS,0)) > 0 then                             
  ((sum(isnull(a.OPENKGS,0))+sum(isnull(a.recKGS,0))+ SUM(ISNULL(A.JOBRECKGS,0))+ SUM(ISNULL(A.TRRECKGS,0))+SUM(ISNULL(A.ISSRETKGS,0))+sum(ISNULL(ISSRETKNITKGS,0))  )                            
  -(sum(isnull(a.PRODKGS,0))+sum(ISNULL(ISSKNITKGS,0))+ sum(isnull(a.TRANSKGS,0))+sum(isnull(a.SALESKGS,0))+sum(isnull(a.JWKGS,0))+sum(isnull(a.REJKGS,0))))                            
  else ((sum(isnull(a.recKGS,0))+SUM(ISNULL(A.JOBRECKGS,0))+ SUM(ISNULL(A.TRRECKGS,0))+SUM(ISNULL(A.ISSRETKGS,0))+sum(ISNULL(ISSRETKNITKGS,0)))                            
  -(sum(isnull(a.PRODKGS,0))+sum(ISNULL(ISSKNITKGS,0))+ sum(isnull(a.TRANSKGS,0))+sum(isnull(a.SALESKGS,0))+sum(isnull(a.JWKGS,0))+sum(isnull(a.REJKGS,0))))                             
  END CLOKGS,               
                             
case when sum(isnull(a.OPENVALUE,0)) > 0 then                             
  ((sum(isnull(a.OPENVALUE,0))+sum(isnull(a.recvalue,0))+SUM(ISNULL(A.JOBRECVALUE,0)) + SUM(ISNULL(A.TRRECVALUE,0))+SUM(ISNULL(A.ISSRETVAL,0))+ SUM(ISNULL(ISSRETKINTVAL,0)))-                            
  (sum(isnull(a.PRODVALUE,0))+ SUM(ISNULL(ISSKNITVAL,0))+ sum(isnull(a.TRANSVALUE,0))+sum(isnull(a.SALESVALUE,0))+sum(isnull(a.JWVALUE,0))+sum(isnull(a.REJVALUE,0))))                            
  else ((sum(isnull(a.recvalue,0))+SUM(ISNULL(A.JOBRECVALUE,0)) + SUM(ISNULL(A.TRRECVALUE,0))+SUM(ISNULL(A.ISSRETVAL,0))+SUM(ISNULL(ISSRETKINTVAL,0)))-                            
  (sum(isnull(a.PRODVALUE,0))+ SUM(ISNULL(ISSKNITVAL,0))+ sum(isnull(a.TRANSVALUE,0))+sum(isnull(a.SALESVALUE,0))+sum(isnull(a.JWVALUE,0))+                            
  sum(isnull(a.REJVALUE,0)))) END CLOSEVALUE               
                
                
  FROM (                            
                            
                             
 ---OPENNING                             
 select varcode,LOTNO,LOTDT, sum(opbales-issbales) AS OPENBAL,sum(opboras-issboras) as OPENBOR,                            
 SUM(opgrsKgs)OPGRSKgs,Sum(opTrKgs)OPTRKgs,                             
 sum(opkgs-isskgs) as OPENKGS,sum(OPvalue-issvalue) as OPENVALUE,                            
 0 AS RECBAL,0 AS RECBOR, 0 AS RECGRSKGS, 0 AS RECTRKGS,0 AS RECKGS,0 AS RECVALUE,                            
 0 AS JOBRECBAL,0 AS JOBRECBOR,0 AS JOBRECGRSKGS,0 AS JOBRECTRKGS,0 AS JOBRECKGS,0 AS JOBRECVALUE ,                            
 0 AS TRRECBAL,0 AS TRRECBOR,0 AS TRRECGRSKGS,0 AS TRRECTRKGS,0 AS TRRECKGS,0 AS TRRECVALUE ,                            
 0 AS RTNBAL,0 AS RTNGRSKGS, 0 AS RTNTRKGS,0 AS RTNKGS,0 AS RTNVALUE ,                       
 0 AS PRODBAL,0 AS PRODBOR,0 AS PRODKGS,0 AS PRODVALUE,0 AS SALESBAL,0 AS SALESBOR,0 AS SALESKGS,0 AS SALESVALUE,                            
 0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,0 AS JWBAL,0 AS JWBOR,0 AS JWKGS,0 AS JWVALUE,                             
 0 AS REJBAL, 0 AS REJKGS,                            
 0 AS REJVALUE, 0 AS LOSSBAL,0 AS LOSSKGS,0 AS LOSSVALUE,                         
 0 AS ISSRETBAL,0 AS ISSRETBOR,0 AS ISSRETKGS,0 AS ISSRETVAL  ,0 AS ISSKNITBAL,0 AS ISSKNITBOR,0 AS ISSKNITKGS,0 AS ISSKNITVAL               
   ,0 ISSRETKNITBAL ,0 ISSRETKNITBOR,0 ISSRETKNITKGS,0 ISSRETKINTVAL   ,0 as CTRECBAL, 0 AS CTRECBOR,0  AS CTRECGRSKGS,0 AS CTRECTRKGS,0 AS CTRECKGS ,0 AS CTRECVALUE                  
  from (                            
                             
 select b.varcode,B.LOTNO,B.LOTDT,                    
 case when b.bblflg = 'B' then sum(isnull(b.bales,0)) else 0 end as opbales,                            
 Case when b.bblflg   IN ('R','H') then sum(isnull(b.bales,0)) else 0 end as opboras,                             
 sum(isnull(b.GRSWGT,0)) as opGrskgs,sum(isnull(b.TAREWT,0)) as opTrkgs,sum(isnull(b.FAVAOURABLEWGT,0)) as opkgs,                            
 0 as issbales,0 as issboras,0 as isskgs,sum(isnull(b.FAVAOURABLEWGT,0)) *cast(b.ratekg as decimal (12,4)) as opvalue,                            
 0 as recvalue,                            
 0 as issvalue from rm_lot b where B.DIVCODE=@DIVCODE AND B.lotyear IN (@TYEAR)                             
 AND (b.lotdt < @FROMDATE or UPPER(OPFLG)='Y')and                            
 BALES >0   and (B.rejflg='N' or B.rejdt>=@YFDATE) group by b.varcode,                            
 b.LOTNO , b.LOTDT, b.bblflg, b.ratekg                            
 Union                            
                     
 select b.varcode,B.LOTNO,B.LOTDT,0 as opbales,0 as opboras,0 as opGrskgs, 0 as opTrkgs,0 as opkgs,                            
 /*                    
 case when b.bblflg = 'B' then count(c.baleno)  else 0 end as issbales,                    
 case when b.bblflg   IN ('R','H') then count(c.baleno) else 0 end as issboras,                    
 */                    
                    
 case when (b.bblflg = 'B' AND C.ISSUED='Y') then count(c.baleno)  else 0 end as issbales,                    
 case when b.bblflg   IN ('R','H') then count(c.baleno) else 0 end as issboras,                    
                      
 SUM(ISNULL(C.isskgs,0)) as isskgs ,0 as opvalue,0 as recvalue,                            
 SUM(ISNULL(C.isskgs,0)) * B.RATEKG as issvalue                            
 from rm_lot b,rm_issb c where                     
 --C.ISSUED='Y' AND                     
 B.DIVCODE=C.DIVCODE                             
 AND B.DIVCODE=@DIVCODE AND (b.lotdt < @FROMDATE                            
 OR UPPER(OPFLG)='Y')  and docdt between @YFDATE and @YLDATE  and b.lotno=c.lotno and                            
 B.lotyear in (@TYEAR) and b.lotdt=c.lotdt and b.lottype=c.lottype                             
 and b.catcd=c.catcd  and docdt <@FROMDATE                            
group by b.varcode,B.LOTNO,B.LOTDT,b.bblflg,b.ratekg ,C.issued                       
 Union                            
                         
 --select b.varcode,B.LOTNO,B.LOTDT,case when b.bblflg = 'B' then count(c.baleno)  else 0 end as opbales,case when b.bblflg   IN ('R','H') then count(c.baleno) else 0 end as opboras,                        
 --SUM(ISNULL(C.isskgs,0)) as  opGrskgs, 0 as opTrkgs,SUM(ISNULL(C.isskgs,0)) as opkgs,                        
 --0 as issbales,0 as issboras,0 as isskgs,SUM(ISNULL(C.isskgs,0)) * B.RATEKG as opvalue, 0 as recvalue,0 as Issvalue           
    select b.varcode,B.LOTNO,B.LOTDT,case when (b.bblflg = 'B'  AND C.ISSUED='Y') then count(isnull(c.BALENO ,0)) else 0 end as opbales,                        
 Case when( b.bblflg   IN ('R','H')  AND C.ISSUED='Y') then count(isnull(c.BALENO ,0)) else 0 end as opboras, sum(isnull(c.ISSKGS,0)) as opGrskgs,            
 sum(isnull(c.ISSKGS,0)) as opTrkgs,sum(isnull(c.ISSKGS,0)) as opkgs,                        
 0 as issbales,0 as issboras,0 as isskgs,sum(isnull(c.ISSKGS,0)) *cast(b.ratekg as decimal (12,4)) as opvalue,0 as recvalue,                        
 0 as issvalue                            
 from rm_lot b,rm_issRtnb c                         
 where  B.DIVCODE=C.DIVCODE AND B.DIVCODE=@DIVCODE AND (b.lotdt < @FROMDATE OR UPPER(OPFLG)='Y')                          
 and docdt between @YFDATE and  @YLDATE   and b.lotno=c.lotno AND B.lotyear in (@TYEAR)                         
 and b.lotdt=c.lotdt and b.lottype=c.lottype and b.catcd=c.catcd  and docdt <@FROMDATE                           
 group by b.varcode,B.LOTNO,B.LOTDT,b.bblflg,b.ratekg  ,c.ISSUED                    
                             
 Union                            
 select VARCODE,LOTNO,Lotdt,0 as opbales,0 as opboras,0 as opGrskgs, 0 as opTrkgs,0 as opkgs,                            
 sum(issBales) AS issbales,0 as issboras,round((sum(isnull(isskgs,0))),3) AS isskgs,0 as opvalue,0 as recvalue,                            
 (sum(isnull(issvalue,0))) as issvalue FROM(SELECT E.VARCODE, e.LOTNO,e.lotdt,0 as opbales,                            
 0 as opboras,0 as opkgs,count(*) as issbales,0 as issboras,                            
 round((sum(isnull(e.FAVAOURABLEWGT,0))),3) AS isskgs,SUM(ISNULL(e.netwt,0)) * B.RATEKG as issvalue                             
 from rm_bale e,rm_lot b where (e.rejflg='Y' and e.rejdt<@FROMDATE) AND e.DIVCODE=@DIVCODE           
 and e.lotno=b.lotno  and e.lotdt=b.lotdt and e.lottype=b.lottype                             
 and e.catcd=b.catcd AND Status='RJ' group by e.varcode,e.LOTNO,e.lotdt,baleno,B.RATEKG)s                             
 GROUP BY  lotno,lotdt, VARCODE ) A GROUP BY A.VARCODE,A.LOTNO,A.LOTDT                             
 HAVING sum(opbales-issbales)>0 OR sum(opboras-issboras) > 0 --OR sum(opKGS-issKGS)>0           
                             
 Union                            
 --RECEIPTS                               
 SELECT  VARCODE, LOTNO,LOTDT,0 AS OPENBAL,0 AS OPENBOR,0 AS OPGRSKgs, 0 AS OPTRKgs, 0 AS OPENKGS,0 AS OPENVALUE,                            
 case when b.bblflg = 'B' then sum(isnull(b.bales,0)) else 0 end as RECBAL,                            
 case when b.bblflg   IN ('R','H') then sum(isnull(b.bales,0)) else 0 end as RECBOR,            
 Case when 'N' = 'N' then sum(isnull(b.GRSWGT,0)) else 0 end as RECGRSKGS,                             
 Case when 'N' = 'N' then sum(isnull(b.TAREWT,0)) else 0 end as RECTRKGS,                            
 Case when 'N' = 'N' then sum(isnull(b.FAVAOURABLEWGT,0)) else 0 end as RECKGS,                            
 SUM(B.FAVAOURABLEWGT)*RATEKG as RECVALUE,                             
 0 AS JOBRECBAL,0 AS JOBRECBOR,0 AS JOBRECGRSKGS,0 AS JOBRECTRKGS,0 AS JOBRECKGS,0 AS JOBRECVALUE ,   0 AS TRRECBAL,0 AS TRRECBOR,0 AS TRRECGRSKGS,0 AS TRRECTRKGS,0 AS TRRECKGS,0 AS TRRECVALUE ,                            
 0 AS RTNBAL,0 AS RTNGRSKGS,0 AS RTNTRKGS,0 AS RTNKGS,0 AS RTNVALUE ,                            
 0 AS PRODBAL,0 AS PRODBOR,0 AS PRODKGS,0 AS PRODVALUE,                            
 0 AS SALESBAL,0 AS SALESBOR,0 AS SALESKGS,0 AS SALESVALUE,                            
 0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,                            
 0 AS JWBAL,0 AS JWBOR,0 AS JWKGS,0 AS JWVALUE,                            
 0 AS REJBAL, 0 AS REJKGS, 0 AS REJVALUE,                             
 0 AS LOSSBAL,0 AS LOSSKGS,0 AS LOSSVALUE,0 AS ISSRETBAL,0 AS ISSRETBOR,0 AS ISSRETKGS,0 AS ISSRETVAL               
 ,0 AS ISSKNITBAL,0 AS ISSKNITBOR,0 AS ISSKNITKGS,0 AS ISSKNITVAL               
 ,0 ISSRETKNITBAL ,0 ISSRETKNITBOR,0 ISSRETKNITKGS,0 ISSRETKINTVAL        
   ,0 as CTRECBAL, 0 AS CTRECBOR,0  AS CTRECGRSKGS,0 AS CTRECTRKGS,0 AS CTRECKGS ,0 AS CTRECVALUE              
 from rm_lot b                             
 where B.DIVCODE=@DIVCODE  AND upper(opflg)='N' and (B.rejflg='N' or B.rejdt>@TODATE) and                            
 B.lotyear=@TYEAR AND B.LOTDT BETWEEN @FROMDATE and @TODATE                             
 AND ISNULL(TransferType,'')=''                            
 group by b.varcode,B.LOTNO,                            
 b.LOTDT , b.opflg, b.bblflg, ratekg                            
                             
 UNION                             
 --JOBWORK RECEIPTS                            
 SELECT  VARCODE, LOTNO,LOTDT,0 AS OPENBAL,0 AS OPENBOR,0 AS OPGRSKgs, 0 AS OPTRKgs, 0 AS OPENKGS,0 AS OPENVALUE,                            
 0 AS  RECBAL,0 AS  RECBOR,0 AS  RECGRSKGS, 0 AS RECTRKGS, 0 AS  RECKGS,0 AS  RECVALUE,                             
 case when b.bblflg = 'B' then sum(isnull(b.bales,0)) else 0 end as JOBRECBAL,                            
 case when b.bblflg   IN ('R','H') then sum(isnull(b.bales,0)) else 0 end as JOBRECBOR,                            
 Case when 'N' = 'N' then sum(isnull(b.GRSWGT,0)) else 0 end as JOBRECGRSKGS,                            
 Case when 'N' = 'N' then sum(isnull(b.TAREWT,0)) else 0 end as  JOBRECTRKGS,                            
 Case when 'N' = 'N' then sum(isnull(b.FAVAOURABLEWGT,0)) else 0 end as JOBRECKGS,                            
 SUM(B.FAVAOURABLEWGT)*RATEKG as JOBRECVALUE ,                            
                             
 0 AS TRRECBAL,0 AS TRRECBOR,0 AS TRRECGRSKGS,0 AS TRRECTRKGS,0 AS TRRECKGS,0 AS TRRECVALUE ,                            
 0 AS RTNBAL,0 AS RTNGRSKGS,0 AS RTNTRKGS,0 AS RTNKGS,0 AS RTNVALUE ,    
 0 AS PRODBAL,0 AS PRODBOR,0 AS PRODKGS,0 AS PRODVALUE,                            
 0 AS SALESBAL,0 AS SALESBOR,0 AS SALESKGS,0 AS SALESVALUE,                            
 0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,                            
 0 AS JWBAL,0 AS JWBOR,0 AS JWKGS,0 AS JWVALUE,                            
 0 AS REJBAL, 0 AS REJKGS, 0 AS REJVALUE,                             
 0 AS LOSSBAL,0 AS LOSSKGS,0 AS LOSSVALUE,0 AS ISSRETBAL,0 AS ISSRETBOR,0 AS ISSRETKGS,0 AS ISSRETVAL               
 ,0 AS ISSKNITBAL,0 AS ISSKNITBOR,0 AS ISSKNITKGS,0 AS ISSKNITVAL               
 ,0 ISSRETKNITBAL ,0 ISSRETKNITBOR,0 ISSRETKNITKGS,0 ISSRETKINTVAL        
   ,0 as CTRECBAL, 0 AS CTRECBOR,0  AS CTRECGRSKGS,0 AS CTRECTRKGS,0 AS CTRECKGS ,0 AS CTRECVALUE              
 from rm_lot b                             
 where B.DIVCODE=@DIVCODE  AND upper(opflg)='N' and (B.rejflg='N' or B.rejdt>@TODATE) and                            
 B.lotyear=@TYEAR AND B.LOTDT BETWEEN @FROMDATE and @TODATE                             
 AND ISNULL(TransferType,'') in ('JJ' ,'JR')                           
 group by b.varcode,B.LOTNO,                
 b.LOTDT , b.opflg, b.bblflg, ratekg                            
                             
 UNION

 group by b.varcode,B.LOTNO,B.LOTDT,b.opflg,b.bblflg,b.ratekg,I.ISSTYPE                  
                   
 ) V GROUP BY VARCODE,LOTNO,LOTDT
