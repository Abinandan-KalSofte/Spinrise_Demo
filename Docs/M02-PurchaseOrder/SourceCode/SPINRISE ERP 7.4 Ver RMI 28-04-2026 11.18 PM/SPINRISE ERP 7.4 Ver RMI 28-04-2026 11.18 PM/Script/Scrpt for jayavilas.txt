create PROCEDURE Sp_stk_position_statewise
(
@DIVCODE VARCHAR(25),
@FDATE VARCHAR(25),
@TDATE VARCHAR(25)
)
AS
BEGIN
	--DECLARE @totbale AS NUMERIC(15,2)
	--DECLARE @pentot AS NUMERIC(15,2)
	--DECLARE @Totpos AS NUMERIC(15,2)
	
	

   
   

SELECT SUM(x.issqty) AS rnlots,SUM(x.isskgs) AS rnkgs,--Runlottotal
CASE WHEN SUM(x.issqty)=0 THEN  0 else ((cast(cast(100 as numeric(15,3))/ cast (ISNULL(x3.Runlottotal,0)  as numeric(15,3)) as numeric(15,3)))*SUM(isnull(x.issqty,0))) end AS rnlotper,

SUM(isnull(x.clobales,0)) waitbale,
CASE WHEN SUM(isnull(x.clobales,0))=0 THEN  0 else ((cast(cast(100 as numeric(15,3))/cast (ISNULL(x2.clobales,0)  as numeric(15,3)) as numeric(15,3)))*SUM(isnull(x.clobales,0))) end AS waitper,
CASE WHEN SUM(x.PenOrderQty)=0 THEN 0 ELSE ((cast(cast(100 as numeric(15,3))/cast (ISNULL(x1.PenOrderQty1,0)  as numeric(15,3)) as numeric(15,3)))*SUM(isnull(x.PenOrderQty,0)))
 end AS ordpend,
SUM(x.cloweight) AS cloweight,SUM(x.clovalue) AS clovalue,
x.name AS state,x.state_code,ISNULL(x1.PenOrderQty1,0)  AS a,ISNULL(x1.PenOrderQty1,0)+ISNULL(x2.clobales,0)  AS b,SUM(x.PenOrderQty) ordpenbale,
CASE WHEN (SUM(ISNULL(PenOrderQty,0))+ SUM(ISNULL(x.clobales,0)))=0 THEN 0
else
(cast(100 as numeric(15,3))/cast(ISNULL(x1.PenOrderQty1,0)+ISNULL(x2.clobales,0) as numeric(15,3))*( SUM(cast(ISNULL(x.PenOrderQty,0) as numeric(15,3)))+ SUM(cast(ISNULL(x.clobales,0) as numeric(15,3))) ))  END AS per,
sum(ISNULL(x.PenOrderQty,0))+sum(ISNULL(x.clobales,0))  AS totbale,
(SELECT pp_divmas.DIV_PRINTNAME FROM pp_divmas WHERE divcode=@DIVCODE) AS DIV_PRINTNAME,(SELECT pp_divmas.UNIT_NAME FROM pp_divmas WHERE divcode=@DIVCODE) AS div_UNITNAME


FROM (
SELECT d.name,d.state_code,SUM(k.bales) AS clobales,SUM(WEIGHT) AS cloweight,SUM(VALUE) AS clovalue,0 AS issqty,0 AS isskgs,0 AS PenOrderQty

  FROM stockpositionwise k inner join rm_lot k1 ON k.lotno=k1.lotno AND k.lotdt=k1.lotdt 
INNER JOIN fa_slmas C ON C.slcode=k1.supcd 
 INNER JOIN ig_state D ON C.state_code=D.STATE_CODE
WHERE k1.divcode=@DIVCODE  GROUP BY d.name,d.state_code 
UNION ALL
 SELECT D.name,d.state_code,0 AS clobales,0 AS cloweight,0 AS clovalue,count(baleno) AS issqty,sum(isskgs) AS isskgs,0 AS PenOrderQty
   FROM rm_issb A INNER join rm_lot B ON A.lotno=B.lotno AND A.lotdt=B.lotdt AND A.lottype=B.LOTTYPE AND A.divcode=B.DIVCODE
 INNER JOIN fa_slmas C ON C.slcode=b.supcd 
 INNER JOIN ig_state D ON C.state_code=D.STATE_CODE
  WHERE  A.divcode=@DIVCODE AND docdt BETWEEN @FDATE AND @TDATE GROUP BY d.name,d.state_code 
UNION ALL 
   SELECT D.name,d.state_code,0 AS clobales,0 AS cloweight,0 AS clovalue,0 AS issqty,0 AS isskgs,(SUM(ORDQTY)-SUM(RECQTY)) AS PenOrderQty
   
    FROM rm_cont A INNER JOIN fa_slmas C ON C.slcode=A.supcd 
   INNER JOIN ig_state D ON C.state_code=D.STATE_CODE
   WHERE A.divcode=@DIVCODE AND A.contdt BETWEEN @FDATE AND @TDATE  GROUP BY d.name,d.state_code having (SUM(A.ORDQTY)-SUM(A.RECQTY))>0  ) x 
   LEFT JOIN (SELECT (SUM(ORDQTY)-SUM(isnull(RECQTY,0))) AS PenOrderQty1,a.divcode FROM rm_cont A INNER JOIN fa_slmas C ON C.slcode=A.supcd 
   INNER JOIN ig_state D ON C.state_code=D.STATE_CODE
   WHERE A.divcode=@DIVCODE AND A.contdt BETWEEN @FDATE AND @TDATE GROUP BY a.divcode  having (SUM(A.ORDQTY)-SUM(A.RECQTY))>0 
               )
   x1 ON  x1.divcode=@DIVCODE
   LEFT JOIN (SELECT SUM(k.bales) AS clobales ,divcode
  FROM stockpositionwise k inner join rm_lot k1 ON k.lotno=k1.lotno AND k.lotdt=k1.lotdt 
INNER JOIN fa_slmas C ON C.slcode=k1.supcd 
 INNER JOIN ig_state D ON C.state_code=D.STATE_CODE
WHERE k1.divcode=@DIVCODE GROUP BY divcode ) x2 ON x2.divcode=@DIVCODE 
   LEFT JOIN  ( SELECT count(baleno) AS Runlottotal,a.divcode
   FROM rm_issb A INNER join rm_lot B ON A.lotno=B.lotno AND A.lotdt=B.lotdt AND A.lottype=B.LOTTYPE AND A.divcode=B.DIVCODE
 INNER JOIN fa_slmas C ON C.slcode=b.supcd 
 INNER JOIN ig_state D ON C.state_code=D.STATE_CODE
  WHERE  A.divcode=@DIVCODE AND docdt BETWEEN @FDATE AND @TDATE GROUP BY a.divcode ) x3 
  ON  x3.divcode=@DIVCODE

GROUP BY  x.name,x.state_code,x1.PenOrderQty1,x2.clobales,x3.Runlottotal

	--SET @pentot= ISNULL((SELECT (SUM(ORDQTY)-SUM(isnull(RECQTY,0))) AS PenOrderQty FROM rm_cont A INNER JOIN fa_slmas C ON C.slcode=A.supcd 
 --  INNER JOIN ig_state D ON C.state_code=D.STATE_CODE
 --  WHERE A.divcode=@DIVCODE AND A.contdt BETWEEN @FDATE AND @TDATE   having (SUM(A.ORDQTY)-SUM(A.RECQTY))>0 ),0)
   
--   SET @totbale=ISNULL((SELECT SUM(k.bales) AS clobales
--  FROM stockpositionwise k inner join rm_lot k1 ON k.lotno=k1.lotno AND k.lotdt=k1.lotdt 
--INNER JOIN fa_slmas C ON C.slcode=k1.supcd 
-- INNER JOIN ig_state D ON C.state_code=D.STATE_CODE
--WHERE k1.divcode=@DIVCODE ),0)

--SET @Totpos=isnull(@pentot,0)+isnull(@totbale,0)


END
go


CREATE PROCEDURE Ksp_Supplier_CodeList    
(    
 @divcode VARCHAR(2)    
)    
AS    
BEGIN    
     
Select Distinct slcode,slname,pd.DIV_PRINTNAME,pd.DIV_UNITNAME from fa_slmas,PP_DIVMAS pd    
 WHERE --SLCODE LIKE 'C1%' AND
 pd.DIVCODE=@divcode    
    
END 

go



CREATE PROCEDURE Ksp_Supplier_DetailedList    
(    
 @divcode VARCHAR(2)    
)    
AS    
BEGIN    
     
Select slcode,slname,    
CASE WHEN isnull(a.phone1,'')<>'' THEN (CASE WHEN isnull(a.phone2,'')<>'' THEN (a.phone1+ ' , '+ a.phone2) ELSE (a.phone1)  end) ELSE (CASE WHEN isnull(a.phone2,'')<>'' THEN (a.phone2) ELSE a.phone2  end) end phone    
,isnull(a.add1,'') as add1,isnull(a.add2,'') as add2,isnull(a.add3,'') as add3,isnull(a.city,'') as city,isnull(a.pin,' ')as pin,a.STATE,a.country    
,a.email,a.TINNO,a.TINDT,a.tngst,a.tngstdt,a.cgstdt,a.cgst,a.ecc,b.DIV_PRINTNAME,b.DIV_UNITNAME from fa_slmas a ,PP_DIVMAS b     
WHERE SLCODE LIKE 'CR%' AND DIVCODE=@divcode ORDER BY slcode    
    
    
END    
    
    
go


  
 -- sp_issuesreport_return '01','IS','2019-03-21','1'  
 CREATE Proc sp_issuesreport_return   
 (  
 @Divcode AS varchar(5),  
  @isstype AS VARCHAR(25),  
 @date AS varchar(10),  
 @dcno AS varchar(10)  
  
 )  
  as     
 Select rm.docdt ,a.VARCODE,a.LotNo,a.LotDt,grwt,b.tarewt,isnull(b.moiSture_ET,0) as moiture,b.netwt,isnull(b.IsQty,0)'IsQty',  
 isnull(rm.ActIssKgs,0) 'AIKgs',isnull(rm.ActIssKgs,0)-isnull(b.IsQty,0) 'Excess',     
 rm.Baleno,rm.actisskgs,f.MIXGRPNAME,g.cntname,0 as ordno,h.divname,h.ADD1,h.ADD2,h.ADD3,h.CITY,h.PINCODE,h.HOState,h.PHONE1 , h.EMAIL, h.WEBADDR, h.FAX, V.VarName, h.DIV_PRINTNAME, h.DIV_UNITNAME, h.DIV_LOGO    
 from rm_lot a inner join rm_bale b on a.Divcode = b.Divcode And a.Varcode = b.Varcode And a.LOTNO = b.LOTNO And a.LOTDT = b.LOTDT And a.catcd = b.catcd    
 inner join   rm_issb c  on  b.DIVCODE=c.DIVCODE AND b.LOTNO=c.LOTNO AND b.BALENO=c.BALENO AND b.VARCODE=c.VARCODE AND b.CATCD=c.CATCD and c.lotdt=b.lotdt     
  inner join  rm_issuetype d on c.ISSTYPE=d.issue_code     
  inner join RM_ISSH e  on e.Divcode = C.Divcode And e.docNo = C.docNo And e.DOCDT = C.DOCDT And e.ISSTYPE = C.ISSTYPE   
  --inner join RM_ISSrtnh e  on e.Divcode = rm.Divcode And e.docNo = rm.docNo And e.DOCDT = rm.DOCDT And e.ISSTYPE = rm.ISSTYPE   
  left join RM_MIXGRP f ON  e.DIVCODE=f.DIVCODE AND  e.MIXGRP=f.MIXGRPCD     
 left join RM_COUNT g on e.DIVCODE=g.Divcode and   e.cntcd=g.cntcd  inner join pp_divmas h  on a.DIVCODE=h.DIVCODE     
 inner join rm_var v  on  a.Varcode = V.Varcode And a.catcd = V.catcd   
 INNER JOIN RM_ISSrtnb rm ON rm.LOTNO=a.LOTNO AND rm.lotdt=a.LOTDT AND a.LOTTYPE=rm.LOTTYPE AND c.DOCNO=rm.ISSNO AND c.docdt=rm.ISSDT  
 inner join RM_ISSrtnh hd on hd.Divcode = rm.Divcode And hd.docNo = rm.docNo And hd.DOCDT = rm.DOCDT And hd.ISSTYPE = rm.ISSTYPe   
 AND rm.ISSTYPE=c.ISSTYPE   
   
 Where     
 a.divcode =@Divcode and hd.docno =1 AND hd.DOCDT=@date --and  a.lotyear=YEAR(@date)  
  AND hd.ISSTYPE=@isstype 
go
