
create View Vw_RM_PendingFreightBill
as
select a.DIVCODE,Fbillno,a.fbilldt ,a.pjno,a.pjdt ,a.supcd Slcode,b.slname,d.VARNAME ,c.lrno+'/'+c.LORRYNOS LORRYNOS
,e.plname,sum(a.comm_wt) comm_wt ,
sum(a.fbillvalue) fbillvalue,sum(c.FREIGHT) FREIGHT,sum(a.bales) bales,a.PlCode,0 ratekg from rm_lot a 
inner join fa_slmas b on a.supcd=b.slcode 
inner join rm_arrival c on a.ARRNO =c.ARRNO and a.arrdt=c.ARRDATE and a.divcode=c.divcode and a.varcode=c.varcode and a.contnum=c.contno and c.CONTDT=a.Contdate
and a.SUPCD=c.supcd  
inner join RM_VAR d on d.VARCODE =a.VARCODE 
left join rm_plant e on e.PLCode=a.PlCode
where isnull(frtentrapp,'N')<>'Y'  and a.PJNO is not null 
group by Fbillno ,a.pjno,a.pjdt ,a.supcd,b.slname,d.VARNAME ,c.lrno+'/'+c.LORRYNOS ,e.plname,a.fbilldt,a.DIVCODE,a.PlCode


go

alter procedure Sp_Freightbill                             
(                        
@Divcode varchar (15),                        
@docno varchar (15),                        
@docdt varchar (15)                        
)                        
as                        
Begin                        
Select a.divcode,d.DIV_LOGO,d.DIV_PRINTNAME,d.DIV_UNITNAME,d.ADD1,d.ADD2,d.ADD3,d.CITY,d.PINCODE,d.STATENAME,                        
d.PHONE1,d.gstinno,d.cinno,d.EMAIL,d.WEBADDR,a.slcode,a.docno as arrno,a.docdt as arrdt,a.billno,a.billdt,a.freightbill_amt,      
a.freightmill_amt,a.freightapp_amt,b.inv_amt,e.slname as carname,b.LRNo as lrno,'' wname,'' as areaname,0 as prno,'' as prmark,      
b.totalkgs as QUANTITY,a.divcode,c.slname,e.slname as agent,e.add1 as aadd1,e.add2 as aadd2,e.add3 as aadd3,e.city as acity,    
e.state as astate,e.country as acountry,e.pin as apin,e.phone1 as aphone,e.email as aemail,e.gstinno as agstno,    
e.gststatecode as agststcode,a.FirstFR_AppUser,a.FirstFR_AppDate,a.SecondFR_AppUser,a.SecondFR_AppDate,a.createdby,a.CreatedDate,    
'' as contname,a.Total_amt,l.BILLNO,l.BILLDT,b.invoiceno,b.invoicedt,b.bales,f.PLName as plant,b.cgstper,b.sgstper,b.igstper,b.cgstamt,    
b.sgstamt,b.igstamt,b.tdsper,b.tdsamt,a.roff_amt,a.value --f.VARNAME as varname,b.contno,b.contdt,b.netwt,b.ratecy,a.LORRYNOS as lorrynos,         
                        
From RM_FreightBillHD a                        
Inner Join RM_FreightBillDT b on b.divcode = a.divcode and b.docno = a.docno and b.docdt = a.docdt and                        
b.docno = a.docno and b.docdt = a.docdt                        
inner Join fa_slmas c on c.slcode=a.slcode                        
inner Join PP_DIVMAS d on d.DIVCODE=a.divcode                        
inner Join fa_slmas e on e.slcode=a.slcode                        
inner Join rm_plant f on f.PLCode=b.PLCode                   
--LEFT join po_car p ON p.carcode=a.carcode              
left join  po_car q ON q.Slcode=c.slcode          
left join rm_lot l on l.divcode=a.divcode and l.arrno = a.docno and l.arrdt=a.docdt --and f.VARCODE= l.varcode           
--l.fbillno = a.fbillno and l.fbilldt=a.fbilldt                        
WHERE a.docno=@docno AND a.docdt=@docdt AND a.divcode=@divcode           
                     
End
