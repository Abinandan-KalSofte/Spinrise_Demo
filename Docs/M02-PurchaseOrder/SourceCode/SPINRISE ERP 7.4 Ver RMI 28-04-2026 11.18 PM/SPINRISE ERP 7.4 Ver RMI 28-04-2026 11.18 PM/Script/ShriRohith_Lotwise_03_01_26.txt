    
CREATE Procedure ksp_lotwisenew                                                                       
@divcode varchar(2),                                                                                              
@FromDate AS varchar(15),                                                                                                 
@ToDate AS Varchar(15),                    
@lotyear AS Varchar(10),              
@Flot varchar(15),      
@Tlot varchar(15)                                                                                            
As                          
begin                        
--if @MixNo='A'               
Begin                        
select distinct                       
--docno,                      
docdt,            
varcode,catcd,cntname ,catname,            
lotno,            
sum(bale) as bale,                        
unit ,round(sum(isnull(netkgs,0)),3)netkg                      
,sum(isnull(quintal,0)) ratequin,                        
sum(isnull(Amount,0)) AS amount,avg(ratekg) as avgrate,plotno,                      
round(avg(ratekg*355.616),0)as ratecandy,                         
mixgrpcd,cntcode,DIV_PRINTNAME,DIV_UNITNAME,mixgrp from (                      
SELECT c.DOCDT, a.CNTCODE AS CNTCD, d.mixgrpcd, a.VARCODE, a.LOTNO, COUNT(b.BALENO) AS bale,                       
(CASE WHEN a.bblflg = 'R' THEN 'Borah' ELSE 'Bales' END) AS Unit, SUM(ISNULL(b.ISSKGS, 0)) AS 'netkgs', a.PLOTNO, AVG(a.RATEKG) AS 'quintal',                       
SUM(ISNULL(b.ISSKGS, 0)) * a.RATEKG AS AMOUNT, a.RATEKG, ROUND(a.RATEKG * 355.616, 0) AS ratecandy, ISNULL(c.DOCNO, 0) AS docno,                       
a.catcd,f.CATNAME ,p.DIV_PRINTNAME,p.DIV_UNITNAME, c.cntcd AS cntcode  ,d.cntname,c.mixgrp                    
FROM dbo.RM_BALE AS e INNER JOIN                      
dbo.RM_ISSB AS b ON e.DIVCODE = b.DIVCODE AND e.lottype = b.LOTTYPE AND e.LOTNO = b.LOTNO AND e.CATCD = b.CATCD AND                       
e.BALENO = b.BALENO and e.varcode = b.varcode  INNER JOIN                      
dbo.RM_LOT AS a ON e.lotdt = a.LOTDT AND b.DIVCODE = a.DIVCODE AND b.CATCD = a.CATCD AND b.LOTNO = a.LOTNO AND                       
b.lotdt = a.LOTDT and e.varcode = a.VARCODE INNER JOIN                      
dbo.RM_ISSH AS c ON b.DIVCODE = c.DIVCODE AND b.ISSTYPE = c.ISSTYPE AND b.DOCDT = c.DOCDT AND b.DOCNO = c.DOCNO                       
INNER JOIN dbo.RM_CAT as f ON a.CATCD=f.CATCD                      
inner join PP_DIVMAS p on p.DIVCODE=a.DIVCODE                       
LEFT OUTER JOIN dbo.RM_COUNT AS d ON c.cntcd = d.cntcd                      
WHERE (b.LOTTYPE = 'A') AND (a.DIVCODE = @divcode) AND (a.REJFLG = 'N') -- AND (b.issued = 'Y')   
AND (a.LOTYEAR = @lotyear)                      
and c.docdt  between @Fromdate  and @Todate                    
GROUP BY d.mixgrpcd, a.BBLFLG, a.RATEKG, a.RATECY, a.VARCODE, a.LOTNO, c.DOCDT, a.CNTCODE, a.PLOTNO, c.DOCNO, a.CATCD,f.CATNAME, c.cntcd,DIV_PRINTNAME,DIV_UNITNAME ,d.cntname,c.mixgrp                        
) x   where x.LOTNO between @Flot and @Tlot                               
group by varcode,lotno,DOCDT,unit,plotno,                      
--docno,                      
catcd,catname,mixgrpcd,cntcode,DIV_PRINTNAME,DIV_UNITNAME,cntname,mixgrp                         
order by cntcode              
                       
End               
                
END
