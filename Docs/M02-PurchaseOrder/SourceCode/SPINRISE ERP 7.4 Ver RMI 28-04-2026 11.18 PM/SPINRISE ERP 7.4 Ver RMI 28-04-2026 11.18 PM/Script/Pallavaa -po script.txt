(((round(ISNULL(ORDKGS,0.00)*ISNULL(RATEKG,0.00),0))*a.CASHDISPER/100)+    
ISNULL(ORDKGS,0.00)*    
isnull((select dbo.Cont_DiscountAmount(a.divcode,a.contno,@OrdDate,a.varcode) ),0)) ,0) Disc_value,    
    
round((ISNULL(ORDKGS,0.00)*ISNULL(RATEKG,0.00)),0)    
-((round((ISNULL(ORDKGS,0.00)*ISNULL(RATEKG,0.00) ),0)*a.CASHDISPER/100)+    
ISNULL(ORDKGS,0.00)*    
isnull((select dbo.Cont_DiscountAmount(a.divcode,a.contno,@OrdDate,a.varcode) ),0)) NetTotal    
              
                   
FROM SRICHERAN..RM_CONT A                                                                         
LEFT OUTER JOIN SRICHERAN..RM_BILLADD B ON  A.billadd = B.slcode                                           
LEFT OUTER JOIN SRICHERAN..RM_deladd C ON  A.deladd = C.slcode                                                                        
INNER JOIN SRICHERAN..RM_AREA D ON D.AREACODE=A.AREACODE                                                                 
INNER JOIN SRICHERAN..RM_VAR E ON A.VARCODE = E.VARCODE                                                                       
INNER JOIN SRICHERAN..PP_DIVMAS F ON A.DIVCODE = F.DIVCODE                                                                        
INNER JOIN SRICHERAN..FA_SLMAS G ON A.SUPCD = G.slcode                                                        
left join SRICHERAN..rm_season rs on rs.SEACODE=a.SEASON                                                                        
LEFT OUTER JOIN SRICHERAN..FA_SLMAS_1 G_1 ON A.BRKCD = G_1.slcode                                                                        
LEFT OUTER JOIN SRICHERAN..RM_SAMPLE H ON A.DIVCODE = H.DIVCODE AND A.SNO = H.SNO                                                           
AND A.SUPCD = H.SUPCODE --AND a.contdt  =h.date                                                              
and A.VARCODE = H.VARCODE                                                                        
LEFT OUTER JOIN SRICHERAN..rm_pmode I ON A.PAYMODE = I.paymode                                                                          
LEFT OUTER JOIN SRICHERAN..IG_STATE J ON J.STATE_CODE=B.StateCode                                                                  
LEFT OUTER JOIN SRICHERAN..IG_STATE K ON K.STATE_CODE=C.StateCode                                           
LEFT OUTER JOIN SRICHERAN..RM_RATEUNIT R ON A.rateunit=R.UNITNAME                                                               
WHERE F.DIVCODE = @Divcode AND A.CONTDT=@OrdDate AND A.CONTNO = @OrdNo AND A.DIVCODE = @Divcode                           
                                               
END
