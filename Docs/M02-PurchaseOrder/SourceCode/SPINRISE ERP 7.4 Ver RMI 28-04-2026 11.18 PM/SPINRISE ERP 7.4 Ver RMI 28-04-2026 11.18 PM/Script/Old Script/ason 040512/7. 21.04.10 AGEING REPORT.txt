CREATE PROCEDURE SP_RMI_STKAGEING 
@DIVCODE AS VARCHAR(3),@YFDATE AS VARCHAR(15),@LOTDT AS VARCHAR(15),@LOTYEAR VARCHAR(5) AS 
 SELECT LOTDT,varcode ,SUM(rbal)-SUM(ibal) AS stkbal,CAST(SUM(rval)-SUM(ival) AS NUMERIC(25,2)) AS stkVal 
 INTO TMP_RMI_STKAGEING FROM (
 SELECT LOTDT,varcode, SUM(bales) AS rbal,SUM(rvalue) AS rval,0 AS ibal,0 AS ival  FROM (	
 SELECT LOTNO,lotdt,varcode,SUM(bales)AS bales,sum(netwt) AS netwt,sum(netwt)*SUM(ratekg) AS rValue, 
 SUM(ratekg) AS ratekg FROM (
 SELECT LOTNO,lotdt,varcode,SUM(bales) AS bales,SUM(FAVAOURABLEWGT) AS netwt, 0 AS ratekg FROM RM_LOT
 WHERE DIVCODE=@DIVCODE AND LOTYEAR=@LOTYEAR AND LOTDT<=@LOTDT GROUP BY LOTNO,lotdt,varcode
 Union All
 SELECT LOTNO,lotdt,varcode,0 AS bales,0 AS netwt, ratekg AS ratekg FROM RM_lot
 WHERE DIVCODE=@DIVCODE AND LOTYEAR=@LOTYEAR AND LOTDT<=@LOTDT GROUP BY lotno,lotdt,varcode,ratekg
 ) y  GROUP BY LOTNO,lotdt,varcode
 ) x GROUP BY LOTDT,varcode 
 Union ALL
 SELECT LOTDT,varcode,0 AS rbal, 0 AS rval, SUM(bales) AS ibal,SUM(rvalue) AS ival FROM (
 SELECT LOTNO,lotdt,varcode,SUM(bales)AS bales,sum(netwt) AS netwt,sum(netwt)*SUM(ratekg) AS rValue, 
 SUM(ratekg) AS ratekg FROM (
 SELECT LOTNO,lotdt,varcode,COUNT(baleno) AS bales,SUM(isskgs) AS netwt, 0 AS ratekg FROM RM_issb
 WHERE DIVCODE=@DIVCODE AND docdt BETWEEN @YFDATE AND @LOTDT  GROUP BY LOTNO,lotdt,varcode
 Union All
 SELECT LOTNO,lotdt,varcode,0 AS bales,0 AS netwt, ratekg AS ratekg FROM RM_lot
 WHERE DIVCODE=@DIVCODE AND LOTYEAR=@LOTYEAR AND LOTDT<=@LOTDT GROUP BY LOTNO,lotdt,varcode,ratekg
 ) y GROUP BY LOTNO,lotdt,varcode
 ) x GROUP BY LOTDT,varcode 
 ) c GROUP by LOTDT,varcode 





ALTER PROCEDURE SP_RMI_STKAGEING1
@LOTDT AS VARCHAR(15) AS 
SELECT A.VARCODE,(SELECT ISNULL(SUM(STKBAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-3,@LOTDT) AND @LOTDT) AS ONEto3bal
,(SELECT ISNULL(SUM(STKVAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-3,@LOTDT) AND @LOTDT) AS ONEto3val
,(SELECT ISNULL(SUM(STKBAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-6,@LOTDT) AND DATEADD(M,-3,@LOTDT)-1) AS THREETO6BAL
,(SELECT ISNULL(SUM(STKVAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-6,@LOTDT) AND DATEADD(M,-3,@LOTDT)-1) AS THREETO6VAL
,(SELECT ISNULL(SUM(STKBAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-9,@LOTDT) AND DATEADD(M,-6,@LOTDT)-1) AS SIXTO9BAL
,(SELECT ISNULL(SUM(STKVAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-9,@LOTDT) AND DATEADD(M,-6,@LOTDT)-1) AS SIXTO9VAL
,(SELECT ISNULL(SUM(STKBAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-12,@LOTDT) AND DATEADD(M,-9,@LOTDT)-1) AS NINETO12BAL
,(SELECT ISNULL(SUM(STKVAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-12,@LOTDT) AND DATEADD(M,-9,@LOTDT)-1) AS NINETO12VAL
,(SELECT ISNULL(SUM(STKBAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT <=DATEADD(M,-12,@LOTDT)-1) AS MORE12BAL
,(SELECT ISNULL(SUM(STKVAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT <=DATEADD(M,-12,@LOTDT)-1) AS MORE12VAL
 FROM TMP_RMI_STKAGEING A 
GROUP BY A.VARCODE
GO
