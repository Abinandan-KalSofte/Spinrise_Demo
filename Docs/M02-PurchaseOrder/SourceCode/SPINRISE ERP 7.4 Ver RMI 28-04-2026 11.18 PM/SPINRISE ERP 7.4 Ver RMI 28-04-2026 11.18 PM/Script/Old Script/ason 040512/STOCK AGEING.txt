SELECT LOTDT,varcode ,SUM(rbal)-SUM(ibal) AS stkbal,CAST(SUM(rval)-SUM(ival) AS NUMERIC(25,2)) AS stkVal 
 FROM (
 SELECT LOTDT,varcode, SUM(bales) AS rbal,SUM(rvalue) AS rval,0 AS ibal,0 AS ival FROM (	
 SELECT LOTNO,lotdt,varcode,SUM(bales)AS bales,sum(netwt) AS netwt,sum(netwt)*SUM(ratekg) AS rValue, 
 SUM(ratekg) AS ratekg FROM (
 SELECT LOTNO,lotdt,varcode,SUM(bales) AS bales,SUM(FAVAOURABLEWGT) AS netwt, 0 AS ratekg FROM RM_LOT
 WHERE DIVCODE='01' AND LOTYEAR='2010' AND LOTDT<='2010-04-01' GROUP BY LOTNO,lotdt,varcode
 Union All
 SELECT LOTNO,lotdt,varcode,0 AS bales,0 AS netwt, ratekg AS ratekg FROM RM_lot
 WHERE DIVCODE='01' AND LOTYEAR='2010' AND LOTDT<='2010-04-01' GROUP BY lotno,lotdt,varcode,ratekg
 ) y  GROUP BY LOTNO,lotdt,varcode
 ) x GROUP BY LOTDT,varcode
 
 Union ALL
 
 SELECT LOTDT,varcode,0 AS rbal, 0 AS rval, SUM(bales) AS ibal,SUM(rvalue) AS ival FROM (
 SELECT LOTNO,lotdt,varcode,SUM(bales)AS bales,sum(netwt) AS netwt,sum(netwt)*SUM(ratekg) AS rValue, 
 SUM(ratekg) AS ratekg FROM (
 SELECT LOTNO,lotdt,varcode,COUNT(baleno) AS bales,SUM(isskgs) AS netwt, 0 AS ratekg FROM RM_issb
 WHERE DIVCODE='01' AND docdt BETWEEN '2010-04-01' AND '2010-04-01'  GROUP BY LOTNO,lotdt,varcode
 Union All
 SELECT LOTNO,lotdt,varcode,0 AS bales,0 AS netwt, ratekg AS ratekg FROM RM_lot
 WHERE DIVCODE='01' AND LOTYEAR='2010' AND LOTDT<='2010-04-01' GROUP BY LOTNO,lotdt,varcode,ratekg
 ) y GROUP BY LOTNO,lotdt,varcode
 ) x GROUP BY LOTDT,varcode 
 ) c GROUP by LOTDT,varcode ORDER BY c.LOTDT DESC


SELECT A.VARCODE,(SELECT ISNULL(SUM(STKBAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-3,'2010-04-01') AND '2010-04-01') AS ONEto3bal
,(SELECT SUM(STKVAL) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-3,'2010-04-01') AND '2010-04-01') AS ONEto3val
,(SELECT ISNULL(SUM(STKBAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-6,'2010-04-01') AND DATEADD(M,-3,'2010-04-01')-1) AS THREETO6BAL
,(SELECT SUM(STKVAL) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-6,'2010-04-01') AND DATEADD(M,-3,'2010-04-01')-1) AS THREETO6VAL
,(SELECT ISNULL(SUM(STKBAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-9,'2010-04-01') AND DATEADD(M,-6,'2010-04-01')-1) AS SIXTO9BAL
,(SELECT SUM(STKVAL) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-9,'2010-04-01') AND DATEADD(M,-6,'2010-04-01')-1) AS SIXTO9VAL
,(SELECT ISNULL(SUM(STKBAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-12,'2010-04-01') AND DATEADD(M,-9,'2010-04-01')-1) AS NINETO12BAL
,(SELECT SUM(STKVAL) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT BETWEEN DATEADD(M,-12,'2010-04-01') AND DATEADD(M,-9,'2010-04-01')-1) AS NINETO12VAL
,(SELECT ISNULL(SUM(STKBAL),0) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT <=DATEADD(M,-12,'2010-04-01')-1) AS MORE12BAL
,(SELECT SUM(STKVAL) FROM TMP_RMI_STKAGEING B 
WHERE A.VARCODE=B.VARCODE AND B.LOTDT <=DATEADD(M,-12,'2010-04-01')-1) AS MORE12VAL
 FROM TMP_RMI_STKAGEING A
GROUP BY A.VARCODE
