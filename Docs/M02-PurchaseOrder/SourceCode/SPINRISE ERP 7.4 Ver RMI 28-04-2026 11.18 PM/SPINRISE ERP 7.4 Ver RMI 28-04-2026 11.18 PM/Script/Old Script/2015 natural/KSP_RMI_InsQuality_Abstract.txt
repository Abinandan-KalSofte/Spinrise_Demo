CREATE PROC KSP_RMI_InsQuality_Abstract    
(    
 @Divcode VARCHAR(2),    
 @FDate VARCHAR(10),    
 @TDate VARCHAR(10)     
)      
AS    
BEGIN    
SELECT A.SUPCD,B.slname 'Supplier Name',A.VARCODE,C.VARNAME,    
A.ORDQTY AS ORDQTY,sum(E.BALES) AS BALES,SUM(D.QUANTITY) AS RECQTY,    
AVG(A.CANDYRATE) 'CANDYRATE',AVG(e.RATEKG)'RATEKG'    
FROM RM_CONT A    
LEFT JOIN FA_SLMAS B ON B.slcode=A.SUPCD    
LEFT JOIN RM_VAR C ON C.VARCODE=A.VARCODE    
LEFT JOIN RM_ARRIVAL D ON D.DIVCODE=A.DIVCODE AND D.CONTNO=A.CONTNO AND D.CONTDT=A.CONTDT      
LEFT JOIN rm_lot e ON e.DIVCODE = D.DIVCODE AND e.ARRNO = D.ARRNO AND e.LOTNO = D.LOTNO AND e.lotdt = D.lotdt     
AND e.lottype = D.lottype    
WHERE    
A.CONTNO IN( SELECT DISTINCT X.CONTNO  FROM RM_ARRIVAL X WHERE X.ARRDATE BETWEEN @FDate AND @TDate AND x.DIVCODE=@Divcode AND a.CONTNO=x.CONTNO AND a.CONTDT=x.CONTDT AND a.DIVCODE=x.DIVCODE)    
and A.DIVCODE=@Divcode AND D.ARRDATE BETWEEN @FDate AND @TDate AND d.DIVCODE=@Divcode    
GROUP BY A.SUPCD,B.slname,A.VARCODE,C.VARNAME , A.ORDQTY  
END
