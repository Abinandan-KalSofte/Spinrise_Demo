alter PROC rm_sam    
(    
 @Divcode VARCHAR(3),      
 @SamNo NUMERIC(10),      
 @FDate VARCHAR(15)      
 )    
AS      
BEGIN      
 SELECT a.DIVCODE,a.SNO,a.DATE,a.SUPCODE, b.slname,b.add1,b.add2,b.add3,b.city,b.pin,b.[state],b.country,b.phone1,b.phone2,a.VARCODE,c.varname,a.immature,a.yellowtouch,      
a.TRASH_PER,
 a.STAPLEN25 ,a.STAPLEN50,a.MICRONAIRE,
a.STRENGTH,a.MCOEFF,remarks,a.avgcount, a.UNIRATIO,a.yarnapp,a.STAPHANDTEST,      
 a.HVITEST,a.moiture,a.tpi,a.csp ,a.tpi ,a.avrstrength , a.corr_strenth , a.spot_for , a.rate ,a.prno ,a.prty_sampno,      
 a.brokercode , a.station_code,POTYPE,GRADE,GTEX,COLOR,D.AREANAME,a.STATE_CODE,STATION_NAME,CROP_YEAR,        
 PD.DIVCODE,PD.DIV_PRINTNAME,PD.DIV_UNITNAME,PD.div_logo,a.REMARKS,pd.ADD2,pd.ADD3,pd.CITY,pd.PINCODE,pd.HOState,pd.HOCountry,pd.PHONE1,pd.PHONE2,    
 pd.PHONE3,pd.FAX,pd.EMAIL,pd.WEBADDR,a.QUANTITY,a.SHORTFIBRE,a.lotno AS supp_lot_no,e.slname AS agent,pd.CUST_ID  
 FROM RM_SAMPLE a    
 inner join fa_slmas b ON a.supcode=b.slcode    
 INNER JOIN rm_var c ON a.varcode=c.varcode     
 LEFT outer JOIN  RM_AREA d ON a.area_code = d.areacode    
 LEFT outer JOIN  fa_slmas e ON   e.slcode = a.brokercode  
 inner join PP_DIVMAS pd ON  PD.DIVCODE =A.DIVCODE      
 WHERE        
 a.date = @FDate and a.sno=@SamNo and a.divcode=@Divcode       
END
