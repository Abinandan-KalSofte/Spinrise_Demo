CREATE proc ksp_Comsumption_RMI  
@Divcode varchar(3),  
@fDate varchar(15),   
@TDate varchar(15),  
@CapCat varchar(15) AS  
  
select * from (  
SELECT g.Consu_DrGlcode ,g.Consu_CrGlcode, isnull(SUM(issval),0)-ISNULL(SUM(RECVAL),0) as value  FROM   (  
  
select distinct m.Catcd,SUM(Value) AS issval,0 as RECVAL FROM(    
select distinct cntcd,b.Catcd,a.lotno,a.lotdt,plotno,Unit=(case when bblflg='B' then 'Bales' else 'Borah' end),
B.varcode,ISSTO,0 as Bales,bblflg,SUM(A.ISSKGS) as Kgs,ratekg, ROUND(avg(ratekg)*355.616,0) as ratecandy,
ratekg*sum(a.isskgs) as Value from rm_issb a,rm_lot b,rm_issh h where    a.docno=h.docno and a.docdt=h.docdt and a.isstype=h.isstype and a.divcode=h.divcode and a.lotno=b.lotno and a.catcd=b.catcd and a.lotdt=b.lotdt and a.divcode=b.divcode AND ISSUED='Y' AND H.divcode=@Divcode 
AND H.DOCDT BETWEEN  @fDate and @TDate group by cntcd,b.Catcd,a.lotno,a.lotdt,plotno,B.varcode,bblflg,ratekg,
ISSTO)m   GROUP BY m.Catcd   
  
) X ,rm_cat g  where x.Catcd = g.Catcd  
GROUP BY g.Consu_DrGlcode,g.Consu_CrGlcode HAVING  isnull(SUM(issval),0)-ISNULL(SUM(RECVAL),0)<>0)x
