alter PROCEDURE Ksp_RMI_ArrivalList_Datewise    
@Divcode  VARCHAR(2),          
@FDate    VARCHAR(10),          
@TDate    VARCHAR(10),
@FSupcd varchar(8),
@Tsupcd varchar(8)
-- Ksp_RMI_ArrivalList_Datewise '01','2013-06-01','2013-07-10'    
AS    
BEGIN    
        
select a.arrdate,a.lotno,a.lotdt,a.plotno,a.quantity,a.lrno,isnull(a.freight,0) as freight,a.netwt,a.prno,    
b.carname,c.Slname , d.areaname, a.LORRYNOS,a.VARCODE,v.VARNAME,e.slname AS Agent_Name    
,pd.DIV_PRINTNAME,pd.DIV_UNITNAME ,a.arrno 
 from rm_arrival a    
 LEFT OUTER JOIN po_car b ON b.CARCODE = a.CARCODE    
 LEFT OUTER JOIN fa_slmas c ON c.Slcode = a.SUPCD    
 LEFT OUTER JOIN rm_area d ON a.areacode=d.areacode    
 LEFT OUTER JOIN RM_VAR V ON v.VARCODE=a.VARCODE    
 LEFT OUTER JOIN fa_slmas e ON e.slcode=a.BRKCD    
 left outer join PP_DIVMAS pd ON a.DIVCODE=pd.DIVCODE    
where a.arrdate Between @FDate and @TDate and a.divcode=@Divcode and  (@FSupcd='A' or a.SUPCD between @FSupcd and @TSupcd)   
END  

go

as 

    select * from (   
select arrdate adate,cast(arrno as varchar) arrno ,Slname, Agent_Name ,lotno,plotno,carname 
, areaname,VARNAME,quantity ,netwt,LORRYNOS ,isnull(freight,0) as freight --,, convert(varchar,a.lotdt,103) lotdt,a.lrno,a.prno,    
   
 from RPTArrivalDetails
 union all
 select arrdate adate,'' arrno,'Sub Total' Slname,'' AS Agent_Name  ,'' lotno,'' plotno,''carname ,
 ''areaname,''VARNAME,SUM(quantity) quantity,SUM(netwt) netwt ,''LORRYNOS ,sum(isnull(a.freight,0)) as freight   --,'' lotdt,'' lrno,'' prno,    
  from RPTArrivalDetails a 
  group by arrdate
  union all
 select '2050-01-01' adate,'' arrno,'Grand Total' Slname,'' AS Agent_Name  ,'' lotno,'' plotno,''carname ,
 ''areaname,''VARNAME,SUM(quantity) quantity,SUM(netwt) netwt ,''LORRYNOS ,sum(isnull(a.freight,0)) as freight   --,'' lotdt,'' lrno,'' prno,    
  from RPTArrivalDetails a  
 )x --order by x.arrdate,lotno desc
