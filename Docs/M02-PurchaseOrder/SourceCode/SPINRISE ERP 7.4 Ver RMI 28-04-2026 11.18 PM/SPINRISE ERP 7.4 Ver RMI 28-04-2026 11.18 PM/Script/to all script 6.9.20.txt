create View Vw_PRDetail
as
	  select  a.divcode,a.prno,a.prdate,b.varcode,b.prsno,b.qtyind,b.qtyindkg,
	  ISNULL(qtyind,0)-ISNULL(qtyord,0) BALQTY,ISNULL(qtyindKG,0)-ISNULL(qtyordKG,0)  BALQTYKG,c.varname  from rm_prh a
inner join rm_prl b on a.prno=b.prno and a.prdate=b.prdate and a.divcode=b.divcode 
inner join rm_var c on c.varcode=b.varcode
where   ISNULL(qtyind,0)-ISNULL(qtyord,0) >0 and ISNULL(qtyindKG,0)-ISNULL(qtyordKG,0) >0  and isnull(FL_Flag,'N')<>'Y'



go
