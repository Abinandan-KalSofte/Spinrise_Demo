create View VW_JBIssueDetails
as
select  a.ISSTYPE ,a.divcode, a.slcode,c.slname ,a.docno,b.docdt,b.varcode, d.VARNAME ,c.brcode,c.City
, sum(b.ISSKGS) ISSKGS ,sum(b.ISSKGS-isnull(b.RecKgs,0)) BalKgs 
from rm_issh a
inner join rm_issb b on a.docno=b.docno and a.docdt=b.docdt and a.divcode=b.divcode
inner join RM_VAR d on d.varcode=b.varcode
inner join fa_slmas c on c.slcode=a.slcode 
where b.ISSKGS-isnull(b.RecKgs,0) >0 
and a.ISSTYPE='JJ'
group by a.divcode, a.slcode,c.slname ,a.docno,b.docdt,b.varcode, d.VARNAME,a.ISSTYPE,c.brcode,c.City
