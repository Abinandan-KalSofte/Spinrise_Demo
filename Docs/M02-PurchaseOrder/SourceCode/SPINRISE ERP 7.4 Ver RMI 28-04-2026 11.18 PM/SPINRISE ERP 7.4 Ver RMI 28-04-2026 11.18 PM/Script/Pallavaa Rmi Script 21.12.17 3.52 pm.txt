CREATE proc constockstatementDenier                   
 @divcode varchar(200),                    
 @fdate varchar(15),                    
 @tdate varchar(15),                    
 @yfate varchar(15),                    
   
 @yldate varchar(15)                    
 as                     
 begin         
 SELECT T.DIVCODE,T.ABBR,T.CATNAME,T.VARCODE,T.VARNAME , T.Unit,T.BALES+T.BORAS AS BALESBORASQTY,T.STOCK,T.VAL,                
 T.VARDESC,(SELECT TOP 1 SHADENAME FROM  IG_SHADEHD WHERE SHADECODE =RTRIM(T.SHADENO)               
 AND DIVCODE in (select items from dbo.Split(@divcode,',') )) AS SHADENAME,                
 '' AS SHADENO,T.Denier,T.LENGTH,LOTNO,PLOTNO,slname,oldvarcode                          
 FROM        
 (                          
  select D.CATCD,D.CATNAME, A.DIVCODE,PP.ABBR, a.varcode,C.VARNAME,case when b.bblflg = 'B' THEN 'Bales'                     
 when b.bblflg  IN ('R','H') THEN 'Boras' END AS Unit,                    
 case when sum(isnull(a.OPENBAL,0)) > 0 then             
  ((sum(isnull(a.OPENBAL,0))+sum(isnull(a.recBAL,0))+sum(isnull(a.ISSRETBAL,0)) )-(sum(isnull(a.PRODBAL,0))+ sum(isnull(a.TRANSBAL,0))+sum(isnull(a.SALESBAL,0))+sum(isnull(a.JWBAL,0))+sum(isnull(a.REJBAL,0))))            
  else             
   ((sum(isnull(a.recBAL,0))+sum(isnull(a.ISSRETBAL,0)))-(sum(isnull(a.PRODBAL,0))+ sum(isnull(a.TRANSBAL,0))+sum(isnull(a.SALESBAL,0))+sum(isnull(a.JWBAL,0)) +sum(isnull(a.REJBAL,0)))) END BALES,                     
 case when sum(isnull(a.OPENBOR,0)) > 0 then ((sum(isnull(a.OPENBOR,0))+sum(isnull(a.recBOR,0))+SUM(ISNULL(a.ISSRETBOR,0)) )-(sum(isnull(a.PRODBOR,0))+ sum(isnull(a.TRANSBOR,0))+sum(isnull(a.SALESBOR,0))+sum(isnull(a.JWBOR,0))))else ((sum(isnull(a.recBOR,
  
    
      
0))+SUM(ISNULL(a.ISSRETBOR,0)))-(sum(isnull(a.PRODBOR,0))            
+ sum(isnull(a.TRANSBOR,0))+sum(isnull(a.SALESBOR,0))+sum(isnull(a.JWBOR,0)))) END AS  BORAS,                    
                     
 case when sum(isnull(a.OPENKGS,0)) > 0 then ((sum(isnull(a.OPENKGS,0))+sum(isnull(a.recKGS,0))+SUM(ISNULL(a.ISSRETKGS,0)))-(sum(isnull(a.PRODKGS,0))+ sum(isnull(a.TRANSKGS,0))+sum(isnull(a.SALESKGS,0))+sum(isnull(a.JWKGS,0))+sum(isnull(a.REJKGS,0))))else
  
    
      
        
          
 ((sum(isnull(a.recKGS,0))+SUM(ISNULL(a.ISSRETKGS,0))) -(            
sum(isnull(a.PRODKGS,0))+ sum(isnull(a.TRANSKGS,0))+sum(isnull(a.SALESKGS,0))+sum(isnull(a.JWKGS,0))+sum(isnull(a.REJKGS,0)))) END STOCK,            
case when sum(isnull(a.OPENVALUE,0)) > 0               
then ((sum(isnull(a.OPENVALUE,0))+sum(isnull(a.recvalue,0))+SUM(isnull(a.ISSRETVAL,0)))-(sum(isnull(a.PRODVALUE,0))+       
sum(isnull(a.TRANSVALUE,0))+sum(isnull(a.SALESVALUE,0))+sum(isnull(a.JWVALUE,0))+sum(isnull(a.REJVALUE,0))))else ((sum(isnull(a.recvalue,0))+SUM(ISNULL(a.ISSRETVAL,0)))-(sum(isnull(a.PRODVALUE,0))+ sum(isnull(a.TRANSVALUE,0))+sum(isnull(a.SALESVALUE,0))+ 
  
    
             
sum(isnull(a.JWVALUE,0))+sum(isnull(a.REJVALUE,0)))) END VAL,                
 C.VARTYPE,CASE WHEN VARTYPE='G' THEN 'Grey' WHEN VARTYPE='D' THEN 'Dyed' ELSE '' End 'Vardesc',                
 SHADENO,Denier,Length,B.LOTNO,B.PLOTNO,B.SUPCD,E.slname,oldvarcode                  
                
 FROM (        
 select divcode,varcode,LOTNO,LOTDT, sum(opbales-issbales) AS OPENBAL,sum(opboras-issboras) as OPENBOR,      
 SUM(opgrsKgs-Isskgs)OPGRSKgs,Sum(opTrKgs)OPTRKgs,       
 sum(opkgs-isskgs) as OPENKGS,sum(OPvalue-issvalue) as OPENVALUE,0 AS RECBAL,0 AS RECBOR,      
 0 AS RECGRSKGS, 0 AS RECTRKGS,0 AS RECKGS,0 AS RECVALUE, 0 AS RTNBAL,0 AS RTNGRSKGS, 0 AS RTNTRKGS,0 AS RTNKGS,0 AS RTNVALUE ,      
 0 AS PRODBAL,0 AS PRODBOR,0 AS PRODKGS,0 AS PRODVALUE,0 AS SALESBAL,0 AS SALESBOR,0 AS SALESKGS,0 AS SALESVALUE,      
 0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,0 AS JWBAL,0 AS JWBOR,0 AS JWKGS,0 AS JWVALUE, 0 AS REJBAL, 0 AS REJKGS,      
 0 AS REJVALUE, 0 AS LOSSBAL,0 AS LOSSKGS,0 AS LOSSVALUE,0 AS ISSRETBAL,0 AS ISSRETBOR, 0 AS ISSRETKGS, 0 AS ISSRETVAL  from (      
       
 select B.DIVCODE,b.varcode,B.LOTNO,B.LOTDT,case when b.bblflg = 'B' then sum(isnull(b.bales,0)) else 0 end as opbales,      
 Case when b.bblflg   IN ('R','H') then sum(isnull(b.bales,0)) else 0 end as opboras, sum(isnull(b.GRSWGT,0)) as opGrskgs,sum(isnull(b.TAREWT,0)) as opTrkgs,sum(isnull(b.FAVAOURABLEWGT,0)) as opkgs,      
 0 as issbales,0 as issboras,0 as isskgs,sum(isnull(b.FAVAOURABLEWGT,0)) *cast(b.ratekg as decimal (12,4)) as opvalue,0 as recvalue,      
 0 as issvalue from rm_lot b where B.DIVCODE in (select items from dbo.Split(@divcode,',') ) AND B.lotyear in (YEAR(@yfate)) AND (b.lotdt < @fdate or UPPER(OPFLG)='Y')and      
 BALES >0   and (B.rejflg='N' or B.rejdt>=@yfate) group by b.varcode,      
 b.LOTNO , b.LOTDT, b.bblflg, b.ratekg,B.DIVCODE      
 UNION      
  select B.DIVCODE,b.varcode,B.LOTNO,B.LOTDT,0 as opbales,0 as opboras,0 as opGrskgs, 0 as opTrkgs,0 as opkgs,      
 case when (b.bblflg = 'B' AND C.ISSUED='Y') then count(c.baleno)  else 0 end as issbales,case when b.bblflg   IN ('R','H') then count      
 (c.baleno) else 0 end as issboras,SUM(ISNULL(C.isskgs,0)) as isskgs ,0 as opvalue,0 as recvalue,SUM(ISNULL(C.isskgs,0)) * B.RATEKG as issvalue      
 from rm_lot b,rm_issb c where  B.DIVCODE=C.DIVCODE AND B.DIVCODE in (select items from dbo.Split(@divcode,',') ) AND (b.lotdt < @fdate      
 OR UPPER(OPFLG)='Y')  and docdt between @yfate and @yldate  and b.lotno=c.lotno and      
 B.lotyear in (YEAR(@yfate)) and b.lotdt=c.lotdt and b.lottype=c.lottype and b.catcd=c.catcd  and docdt <@fdate      
 group by b.varcode,B.LOTNO,B.LOTDT,b.bblflg,b.ratekg,c.issued,B.DIVCODE      
 Union      
      
 select B.DIVCODE,b.varcode,B.LOTNO,B.LOTDT,case when b.bblflg = 'B' then count(c.baleno)  else 0 end as opbales,case when b.bblflg   IN ('R','H') then count(c.baleno) else 0 end as opboras,      
 SUM(ISNULL(C.isskgs,0)) as  opGrskgs, 0 as opTrkgs,SUM(ISNULL(C.isskgs,0)) as opkgs,      
 0 as issbales,0 as issboras,0 as isskgs,SUM(ISNULL(C.isskgs,0)) * B.RATEKG as opvalue, 0 as recvalue,0 as Issvalue       
 from rm_lot b,rm_issRtnb c where C.ISSUED='Y' AND B.DIVCODE=C.DIVCODE AND B.DIVCODE in (select items from dbo.Split(@divcode,',') )AND (b.lotdt < @fdate      
 OR UPPER(OPFLG)='Y')  and docdt between @yfate and @yldate  and b.lotno=c.lotno and      
 B.lotyear in (YEAR(@yfate)) and b.lotdt=c.lotdt and b.lottype=c.lottype and b.catcd=c.catcd  and docdt <@fdate      
 group by b.varcode,B.LOTNO,B.LOTDT,b.bblflg,b.ratekg,B.DIVCODE,B.DIVCODE      
      
 Union      
 select DIVCODE,VARCODE,LOTNO,Lotdt,0 as opbales,0 as opboras,0 as opGrskgs, 0 as opTrkgs,0 as opkgs,sum(issBales) AS issbales,0 as issboras,round((sum(isnull(isskgs,0))),3) AS isskgs,0 as opvalue,0 as recvalue,      
 (sum(isnull(issvalue,0))) as issvalue FROM ( SELECT B.DIVCODE,E.VARCODE, e.LOTNO,e.lotdt,0 as opbales,0 as opboras,0 as opkgs,count(*) as issbales,0 as issboras,      
 round((sum(isnull(e.FAVAOURABLEWGT,0))),3) AS isskgs,SUM(ISNULL(e.netwt,0)) * B.RATEKG as issvalue       
 from rm_bale e,rm_lot b where (e.rejflg='Y' and e.rejdt<@fdate) AND e.DIVCODE in (select items from dbo.Split(@divcode,',') )       
 and e.lotno=b.lotno  and e.lotdt=b.lotdt and e.lottype=b.lottype and e.catcd=b.catcd AND Status='RJ' group by e.varcode,e.LOTNO,e.lotdt,baleno,B.RATEKG,b.DIVCODE)s GROUP BY       
 lotno,lotdt, VARCODE ,DIVCODE) A GROUP BY A.VARCODE,A.LOTNO,A.LOTDT,divcode HAVING sum(opbales-issbales)>0 OR sum(opboras-issboras) > 0 --OR sum(opKGS-issKGS)>0      
      
 Union      
 SELECT  DIVCODE,VARCODE, LOTNO,LOTDT,0 AS OPENBAL,0 AS OPENBOR,0 AS OPGRSKgs, 0 AS OPTRKgs, 0 AS OPENKGS,0 AS OPENVALUE,case when b.bblflg = 'B' then sum(isnull(b.bales,0)) else 0 end as RECBAL,      
 case when b.bblflg   IN ('R','H') then sum(isnull(b.bales,0)) else 0 end as RECBOR,      
 Case when 'N' = 'N' then sum(isnull(b.GRSWGT,0)) else 0 end as RECGRSKGS, Case when 'N' = 'N' then sum(isnull(b.TAREWT,0)) else 0 end as RECTRKGS,      
 Case when 'N' = 'N' then sum(isnull(b.FAVAOURABLEWGT,0)) else 0 end as RECKGS,SUM(B.FAVAOURABLEWGT)*RATEKG as RECVALUE, 0 AS RTNBAL,0 AS RTNGRSKGS,0 AS RTNTRKGS,0 AS RTNKGS,0 AS RTNVALUE ,      
 0 AS PRODBAL,0 AS PRODBOR,0 AS PRODKGS,0 AS PRODVALUE,0 AS SALESBAL,0 AS SALESBOR,0 AS SALESKGS,0 AS SALESVALUE,      
 0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,0 AS JWBAL,0 AS JWBOR,0 AS JWKGS,0 AS JWVALUE, 0 AS REJBAL, 0 AS REJKGS,      
 0 AS REJVALUE, 0 AS LOSSBAL,0 AS LOSSKGS,0 AS LOSSVALUE,      
 0 AS ISSRETBAL,0 AS ISSRETBOR, 0 AS ISSRETKGS, 0 AS ISSRETVAL from rm_lot b where B.DIVCODE in (select items from dbo.Split(@divcode,',') ) AND upper(opflg)='N' and (B.rejflg='N' or B.rejdt>@tdate) and      
 B.lotyear=(YEAR(@yfate))  AND B.LOTDT BETWEEN @fdate and @tdate group by b.varcode,B.LOTNO,      
 b.LOTDT , b.opflg, b.bblflg, ratekg,B.DIVCODE      
  Union      
 SELECT  DIVCODE,VARCODE, LOTNO,LOTDT,0 AS OPENBAL,0 AS OPENBOR,0 AS OPGRSKgs, 0 AS OPTRKgs,0 AS OPENKGS,0 AS OPENVALUE,0 as RECBAL,0 as RECBOR,      
 0 RECGRSKGS,0 AS RECTRKGS,0 as RECKGS,0 as RECVALUE, isnull(sum(bales),0) as RTNBAL,      
 isnull(sum(GRSWGT),0) as RTNGRSKGS, isnull(sum(TAREWT),0) as RTNTRKGS,isnull(sum(FAVAOURABLEWGT),0) as RTNKGS,      
 round((sum(isnull(FAVAOURABLEWGT,0))),3)*cast(ratekg as decimal (12,4))AS RTNVALUE,      
 0 AS PRODBAL,0 AS PRODBOR,0 AS PRODKGS,0 AS PRODVALUE,0 AS SALESBAL,0 AS SALESBOR,0 AS SALESKGS,0 AS SALESVALUE,      
  0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,0 AS JWBAL,0 AS JWBOR,0 AS JWKGS,0 AS JWVALUE, 0 AS REJBAL, 0 AS REJKGS,      
 0 AS REJVALUE, 0 AS LOSSBAL,0 AS LOSSKGS,0 AS LOSSVALUE,       
 0 AS ISSRETBAL,0 AS ISSRETBOR, 0 AS ISSRETKGS, 0 AS ISSRETVAL       
 from rm_lot where divcode in (select items from dbo.Split(@divcode,',') ) and lotyear=(YEAR(@yfate))       
 and lottype='T' and transfertype='JR' and lotdt Between @fdate and @tdate      
 GROUP BY VARCODE,LOTNO,LOTDT,RATEKG,DIVCODE      
  Union      
 SELECT  b.DIVCODE,C.VARCODE,C.LOTNO,C.LOTDT,0 AS OPENBAL,0 AS OPENBOR,0 AS OPGRSKgs, 0 AS OPTRKgs,0 AS OPENKGS,0 AS OPENVALUE,0 AS RECBAL,0 AS RECBOR,      
 0 AS RECGRSKGS,0 AS RECTRKGS,0 AS RECKGS,0 AS RECVALUE, 0 AS RTNBAL,0 AS RTNGRSKGS,0 AS RTNTRKGS,0 AS RTNKGS,0 AS RTNVALUE ,0 AS PRODBAL,0 AS PRODBOR,0 AS PRODKGS,0 AS PRODVALUE,0 AS SALESBAL,0 AS SALESBOR,      
 0 AS SALESKGS,0 AS SALESVALUE,0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,0 AS JWBAL,0 AS JWBOR,      
 0 AS JWKGS,0 AS JWVALUE,0 AS REJBAL,0 AS REJKGS,0 AS REJVALUE , count(C.baleno) AS LOSSBAL,sum(isnull(C.FAVAOURABLEWGT,0)-isnull(C.FAVAOURABLEWGT,0)) LOSSKGS,      
 round((sum(isnull(C.FAVAOURABLEWGT,0))),3)*cast(b.ratekg as decimal (12,4)) AS LOSSVALUE,      
 0 AS ISSRETBAL,0 AS ISSRETBOR, 0 AS ISSRETKGS, 0 AS ISSRETVAL       
 from RM_BALE C,RM_LOT B      
 Where c.LOTNO = b.LOTNO And c.LOTDT = b.LOTDT And c.catcd = b.catcd And c.Divcode = b.Divcode And c.lottype = b.lottype      
 AND C.lotdt Between @fdate and @tdate and C.Issued = 'Y'      
 group by C.VARCODE,C.LOTNO,C.LOTDT,B.RATEKG,b.DIVCODE      
 Union      
 SELECT  DIVCODE,VARCODE, LOTNO,LOTDT,0 AS OPENBAL,0 AS OPENBOR,0 AS OPGRSKgs, 0 AS OPTRKgs,0 AS OPENKGS,0 AS OPENVALUE,0 AS RECBAL,0 AS RECBOR,0 AS RECGRSKGS,0 AS RECTRKGS,0 AS RECKGS,0 AS RECVALUE, 0 AS RTNBAL,0 AS RTNGRSKGS,0 AS RTNTRKGS,0 AS RTNKGS,0
  
     
AS RTNVALUE,

 SUM(SALESBAL) AS SALESBAL,SUM(SALESBOR) AS SALESBOR,SUM(SALESKGS) AS SALESKGS,SUM(SALESVALUE) AS SALESVALUE,      
 SUM(TRANSBAL) AS TRANSBAL,SUM(TRANSBOR) AS TRANSBOR,SUM(TRANSKGS) AS TRANSKGS,SUM(TRANSVALUE) AS TRANSVALUE,      
 SUM(JWBAL) AS JWBAL,SUM(JWBOR) AS JWBOR,SUM(JWKGS) AS JWKGS,SUM(JWVALUE) AS JWVALUE,  0 AS REJBAL, 0 AS REJKGS,0 AS REJVALUE, 0 AS LOSSBAL,0 AS LOSSKGS,0 AS LOSSVALUE,      
 SUM(ISNULL(ISSRETBAL,0))ISSRETBAL,SUM(ISNULL(ISSRETBOR,0))ISSRETBOR,SUM(ISNULL(ISSRETKGS,0))ISSRETKGS, SUM(ISNULL(ISSRETVAL,0)) AS ISSRETVAL

 (       
   select DIVCODE,varcode,LOTNO,LOTDT,ISSTYPE,SUM(ISNULL(Bales,0))PRODBAL,SUM(ISNULL(BORAS,0)) PRODBOR,

 case when b.bblflg   IN ('R','H') then count(c.baleno) else 0 END as BORAS,0 AS ISSKGS ,0 AS ISSVALUE ,0 as SALESBAL,      
 0 as SALESBOR,0 as SALESKGS,0 AS SALESVALUE,0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,

 0 AS ISSRETBAL,0 AS ISSRETBOR, 0 AS ISSRETKGS, 0 AS ISSRETVAL       
 from rm_lot b,rm_issb c,rm_issuetype i where C.ISSUED='Y' AND B.DIVCODE=C.DIVCODE AND B.DIVCODE in (select items from dbo.Split(@divcode,',') ) AND      
 BALES >0  AND b.lotno=c.lotno and b.lotdt=c.lotdt and b.lottype=c.lottype and b.catcd=c.catcd      
 AND I.issue_code=C.ISSTYPE and docdt between @fdate and @tdate and lotyear=(YEAR(@yfate))       
 and (B.rejflg='N' OR B.rejdt>@tdate) AND I.Isstype='P'      
 group by b.varcode,B.LOTNO,B.LOTDT,b.opflg,b.bblflg,b.ratekg,I.ISSTYPE,b.DIVCODE      
 Union       
 select b.DIVCODE,b.varcode,B.LOTNO,B.LOTDT,I.ISSTYPE,0 as BALES,0 AS BORAS,sum(ISNULL(C.Isskgs,0)) as ISSKGS,      
 round((sum(isnull(C.Isskgs,0))),3)*cast(b.ratekg as decimal (12,4)) AS ISSVALUE,0 as SALESBAL,      
 0 as SALESBOR,0 as SALESKGS,0 AS SALESVALUE,0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,      
 0 AS JWBAL,0 AS JWBOR,0 AS JWKGS,0 AS JWVALUE,       
 0 AS ISSRETBAL,0 AS ISSRETBOR, 0 AS ISSRETKGS, 0 AS ISSRETVAL       
 from rm_lot b,rm_issb c,rm_issuetype i where B.DIVCODE=C.DIVCODE AND B.DIVCODE in (select items from dbo.Split(@divcode,',') ) AND      
 BALES >0  AND b.lotno=c.lotno and b.lotdt=c.lotdt and b.lottype=c.lottype and b.catcd=c.catcd      
 AND I.issue_code=C.ISSTYPE and docdt between @fdate and @tdate and lotyear=(YEAR(@yfate))       
 and (B.rejflg='N' OR B.rejdt>@tdate) AND I.Isstype='P'      
 group by b.varcode,B.LOTNO,B.LOTDT,b.opflg,b.bblflg,b.ratekg,I.ISSTYPE,b.DIVCODE      
 )x group by Varcode,LOTNO,LOTDT,ISSTYPE,DIVCODE      
  Union      
 select b.DIVCODE,b.varcode,B.LOTNO,B.LOTDT,I.ISSTYPE,0 as PRODBAL,0 as PRODBOR,0 as PRODKGS,0 AS PRODVALUE,case when b.bblflg = 'B' then count(c.baleno)  else 0 END as SALESBAL,

,    
      
 0 AS JWBAL,0 AS JWBOR,0 AS JWKGS,0 AS JWVALUE,      
 0 AS ISSRETBAL,0 AS ISSRETBOR, 0 AS ISSRETKGS, 0 AS ISSRETVAL

 C.ISSUED='Y' AND B.DIVCODE=C.DIVCODE AND B.DIVCODE in (select items from dbo.Split(@divcode,',') ) AND BALES >0      
 AND b.lotno=c.lotno and b.lotdt=c.lotdt and b.lottype=c.lottype and b.catcd=c.catcd

 group by b.varcode,B.LOTNO,B.LOTDT,b.opflg,b.bblflg,b.ratekg,I.ISSTYPE,b.DIVCODE      
 Union      
 select b.DIVCODE,b.varcode,B.LOTNO,B.LOTDT,I.ISSTYPE,0 as PRODBAL,0 as PRODBOR,0 as PRODKGS,0 AS PRODVALUE,0 as SALESBAL,      
 0 as SALESBOR,0 as SALESKGS,0 AS SALESVALUE,case when b.bblflg = 'B' then count(c.baleno)  else 0 END AS TRANSBAL,case when b.bblflg   IN ('R','H') then count(c.baleno) else 0 END AS TRANSBOR,sum(ISNULL(C.isskgs,0)) AS TRANSKGS,round((sum(isnull(C.isskgs
  
    
,0))),3)*cast(b.ratekg as decimal (12,4)) AS TRANSVALUE,      
 0 AS JWBAL,0 AS JWBOR,0 AS JWKGS,0 AS JWVALUE,      
 0 AS ISSRETBAL,0 AS ISSRETBOR, 0 AS ISSRETKGS, 0 AS ISSRETVAL       
 from rm_lot b,rm_issb c,rm_issuetype i where      
 C.ISSUED='Y' AND B.DIVCODE=C.DIVCODE AND B.DIVCODE in (select items from dbo.Split(@divcode,',') ) AND BALES >0      
 AND b.lotno=c.lotno and b.lotdt=c.lotdt and b.lottype=c.lottype and b.catcd=c.catcd      
 AND I.issue_code=C.ISSTYPE and docdt between @fdate and @tdate and lotyear=(YEAR(@yfate))       
 and (B.rejflg='N' OR B.rejdt>@tdate) AND I.Isstype='T'      
 group by b.varcode,B.LOTNO,B.LOTDT,b.opflg,b.bblflg,b.ratekg,I.ISSTYPE,b.DIVCODE      
      
 Union      
 select b.DIVCODE,b.varcode,B.LOTNO,B.LOTDT,I.ISSTYPE,0 as PRODBAL,0 as PRODBOR,0 as PRODKGS,0 AS PRODVALUE,0 as SALESBAL,      
 0 as SALESBOR,0 as SALESKGS,0 AS SALESVALUE,0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,

  case when b.bblflg   IN ('R','H') then count(c.baleno) else 0 END AS JWBOR,      
 sum(ISNULL(C.isskgs,0)) AS JWKGS,round((sum(isnull(C.isskgs,0))),3)*cast(b.ratekg as decimal (12,4)) AS JWVALUE,      
 0 AS ISSRETBAL,0 AS ISSRETBOR, 0 AS ISSRETKGS, 0 AS ISSRETVAL       
 from rm_lot b,rm_issb c,rm_issuetype i where      
 C.ISSUED='Y' AND B.DIVCODE=C.DIVCODE AND B.DIVCODE in (select items from dbo.Split(@divcode,',') ) AND BALES >0      
 AND b.lotno=c.lotno and b.lotdt=c.lotdt and b.lottype=c.lottype and b.catcd=c.catcd      
 AND I.issue_code=C.ISSTYPE and docdt between @fdate and @tdate and lotyear=(YEAR(@yfate))       
 and (B.rejflg='N' OR B.rejdt>@tdate) AND I.Isstype='J'      
 group by b.varcode,B.LOTNO,B.LOTDT,b.opflg,b.bblflg,b.ratekg,I.ISSTYPE,b.DIVCODE

 select DIVCODE,varcode,LOTNO,LOTDT,'IR' AS ISSTYPE, 0 AS PRODBAL,0 AS PRODBOR,0 AS PRODKGS,0 AS  PRODVALUE,      
 0 as SALESBAL,0 as SALESBOR,0 as SALESKGS,0 AS SALESVALUE,0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,      
 0 AS JWBAL,0 AS JWBOR,0 AS JWKGS,0 AS JWVALUE,      
 SUM(ISNULL(Bales,0)) ISSRETBAL,SUM(ISNULL(BORAS,0))AS ISSRETBOR,SUM(ISNULL(ISSKGS,0)) AS ISSRETKGS,      
 SUM(IsNull(ISSVALUE, 0)) As ISSRETVAL FROM (      
 Select b.DIVCODE,b.varcode,B.LOTNO,B.LOTDT,case when b.bblflg = 'B' then count(c.baleno)  else 0 END as BALES,      
 case when b.bblflg   IN ('R','H') then count(c.baleno) else 0 END as BORAS,0 AS ISSKGS ,0 AS ISSVALUE ,      
 0 as SALESBAL,0 as SALESBOR,0 as SALESKGS,0 AS SALESVALUE,0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,      
 0 AS JWBAL,0 AS JWBOR,0 AS JWKGS,0 AS JWVALUE       
 From rm_lot b,RM_ISSRTNB c where C.ISSUED='Y' AND B.DIVCODE=C.DIVCODE AND B.DIVCODE in (select items from dbo.Split(@divcode,',') ) AND       
 BALES >0  AND b.lotno=c.lotno and b.lotdt=c.lotdt and b.lottype=c.lottype and b.catcd=c.catcd       
 And docdt between @fdate and @tdate and lotyear=(YEAR(@yfate))       
 and (B.rejflg='N' OR B.rejdt>@tdate)      
 group by b.varcode,B.LOTNO,B.LOTDT,b.opflg,b.bblflg,b.ratekg,b.DIVCODE      
 Union      
 Select b.DIVCODE,b.varcode,B.LOTNO,B.LOTDT,0 as BALES,0 AS BORAS,sum(ISNULL(C.Isskgs,0)) as ISSKGS,      
 round((sum(isnull(C.Isskgs,0))),3)*cast(b.ratekg as decimal (12,4)) AS ISSVALUE,0 as SALESBAL,      
 0 as SALESBOR,0 as SALESKGS,0 AS SALESVALUE,0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,      
 0 AS JWBAL,0 AS JWBOR,0 AS JWKGS,0 AS JWVALUE      
 from rm_lot b,RM_ISSRTNB c where B.DIVCODE=C.DIVCODE AND B.DIVCODE in (select items from dbo.Split(@divcode,',') ) AND      
 BALES >0  AND b.lotno=c.lotno and b.lotdt=c.lotdt and b.lottype=c.lottype and b.catcd=c.catcd      
 And docdt between @fdate and @tdate and lotyear=(YEAR(@yfate))       
 and (B.rejflg='N' OR B.rejdt>@tdate)      
 group by b.varcode,B.LOTNO,B.LOTDT,b.opflg,b.bblflg,b.ratekg,b.DIVCODE      
 )IR group by Varcode,LOTNO,LOTDT ,DIVCODE      
 ) X GROUP BY LOTNO,LOTDT,VARCODE,ISSTYPE ,DIVCODE      
       
  Union      
 SELECT  DIVCODE,VARCODE,LOTNO,LOTDT,0 AS OPENBAL,0 AS OPENBOR,0 AS OPGRSKgs, 0 AS OPTRKgs,0 AS OPENKGS,0 AS OPENVALUE,0 AS RECBAL,0 AS RECBOR,      
 0 AS RECGRSKGS,0 AS RECTRKGS,0 AS RECKGS,0 AS RECVALUE, 0 AS RTNBAL,0 AS RTNGRSKGS,0 AS RTNTRKGS,0 AS RTNKGS,0 AS RTNVALUE,0 AS PRODBAL,0 AS PRODBOR,0 AS PRODKGS,0 AS PRODVALUE,0 AS SALESBAL,0 AS SALESBOR,      
 0 AS SALESKGS,0 AS SALESVALUE,0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,0 AS JWBAL,0 AS JWBOR,      
 0 AS JWKGS,0 AS JWVALUE, SUM(REJBAL) AS REJBAL, SUM(REJKGS) AS REJKGS,      
 SUM(REJVALUE) AS REJVALUE,0 AS LOSSBAL,0 AS LOSSKGS,0 AS LOSSVALUE,       
 0 AS ISSRETBAL,0 AS ISSRETBOR, 0 AS ISSRETKGS, 0 AS ISSRETVAL       
 FROM (      
 SELECT b.DIVCODE, C.VARCODE,C.LOTNO,C.LOTDT,0 AS OPENBAL,0 AS OPENBOR,0 AS OPGRSKgs, 0 AS OPTRKgs,0 AS OPENKGS,0 AS OPENVALUE,0 AS RECBAL,0 AS RECBOR,      
 0 AS RECGRSKGS,0 AS RECTRKGS,0 AS RECKGS,0 AS RECVALUE, 0 AS RTNBAL,0 AS RTNGRSKGS,0 AS RTNTRKGS,0 AS RTNKGS,0 AS RTNVALUE,0 AS PRODBAL,0 AS PRODBOR,0 AS PRODKGS,0 AS PRODVALUE,0 AS SALESBAL,0 AS SALESBOR,      
 0 AS SALESKGS,0 AS SALESVALUE,0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,0 AS JWBAL,0 AS JWBOR,      
 0 AS JWKGS,0 AS JWVALUE, COUNT(C.BALENO) AS REJBAL, SUM(C.FAVAOURABLEWGT) AS REJKGS,      
 round((sum(isnull(C.FAVAOURABLEWGT,0))),3)*cast(b.ratekg as decimal (12,4)) AS REJVALUE,0 AS LOSSBAL,0 AS LOSSKGS,0 AS LOSSVALUE      
 FROM RM_BALE C,RM_LOT B      
 Where c.LOTNO = b.LOTNO And c.LOTDT = b.LOTDT And c.catcd = b.catcd And c.Divcode = b.Divcode And c.lottype = b.lottype      
 AND C.rejflg='Y' and C.Status='RJ' and C.rejdt>=@fdate      
 and C.rejdt<=@tdate and lotyear=(YEAR(@yfate))       
 GROUP BY C.LOTNO,C.lotdt,C.VARCODE,B.RATEKG,b.DIVCODE UNION       
 select b.DIVCODE,b.varcode,B.LOTNO,B.LOTDT,0 AS OPENBAL,0 AS OPENBOR,0 AS OPGRSKgs, 0 AS OPTRKgs,0 AS OPENKGS,0 AS OPENVALUE,0 AS RECBAL,0 AS RECBOR,      
 0 AS RECGRSKGS,0 AS RECTRKGS,0 AS RECKGS,0 AS RECVALUE, 0 AS RTNBAL,0 AS RTNGRSKGS,0 AS RTNTRKGS,0 AS RTNKGS,0 AS RTNVALUE,0 AS PRODBAL,0 AS PRODBOR,0 AS PRODKGS,0 AS PRODVALUE,0 AS SALESBAL,0 AS SALESBOR,      
 0 AS SALESKGS,0 AS SALESVALUE,0 AS TRANSBAL,0 AS TRANSBOR,0 AS TRANSKGS,0 AS TRANSVALUE,0 AS JWBAL,0 AS JWBOR,      
 0 AS JWKGS,0 AS JWVALUE,count(c.baleno) as REJBAL,sum(ISNULL(C.isskgs,0)) as REJKGS,round((sum(isnull(C.isskgs,0))),3)*cast(b.ratekg as decimal (12,4)) AS REJVALUE,0 AS LOSSBAL,0 AS LOSSKGS,0 AS LOSSVALUE      
 from rm_lot b,rm_issb c,rm_issuetype i where C.ISSUED='Y' AND B.DIVCODE=C.DIVCODE AND B.DIVCODE in (select items from dbo.Split(@divcode,',') ) AND      
 BALES >0  AND b.lotno=c.lotno and b.lotdt=c.lotdt and b.lottype=c.lottype and b.catcd=c.catcd      
 AND I.issue_code=C.ISSTYPE and docdt between @fdate and @tdate and lotyear=(YEAR(@yfate))       
 and (B.rejflg='N' OR B.rejdt>@tdate) AND I.Isstype='R'      
 group by b.varcode,B.LOTNO,B.LOTDT,b.opflg,b.bblflg,b.ratekg,I.ISSTYPE,b.DIVCODE      
--  group by C.DIVCODE, b.varcode,B.LOTNO,B.LOTDT,b.opflg,b.bblflg,b.ratekg,I.ISSTYPE        
 ) V GROUP BY VARCODE,LOTNO,LOTDT ,DIVCODE      
  ) A    
  inner join RM_LOT B  on A.LOTNO=B.LOTNO AND A.LOTDT=B.LOTDT AND A.VARCODE =B.VARCODE   
  left join RM_VAR C   on A.VARCODE=C.VARCODE and B.CATCD=C.CATCD   
  LEFT JOIN  rm_CAT D on B.CATCD= D.CATCD                   
 left join FA_SLMAS E ON B.SUPCD=E.SLCODE   
 LEFT JOIN  PP_DIVMAS AS PP   ON  PP.DIVCODE=B.DIVCODE and a.DIVCODE=pp.DIVCODE                  
  WHERE                      
                      
    b.divcode  in (select items from dbo.Split(@divcode,','))                
                
   and b.lotyear=(YEAR(@yfate)) --and (b.transfertype='' or b.transfertype='JR')                     
   and b.lotyear=(YEAR(@yfate)) group by A.DIVCODE,PP.ABBR, a.varcode,C.VARNAME,bblflg,                  
   C.VARTYPE,c.SHADENO,c.Denier,c.LENGTH,D.CATCD,D.CATNAME,B.LOTNO,B.PLOTNO,B.SUPCD,E.slname,oldvarcode                    
   )T   ORDER BY T.DIVCODE,T.VARCODE ASC
