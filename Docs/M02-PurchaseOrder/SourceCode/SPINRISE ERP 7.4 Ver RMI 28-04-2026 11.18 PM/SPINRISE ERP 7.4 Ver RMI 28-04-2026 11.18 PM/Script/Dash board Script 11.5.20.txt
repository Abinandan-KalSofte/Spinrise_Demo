create view Vw_RM_OrderQty
as
 select DISTINCT sum(a.ordqty) ordqty , DIVCODE ,CONTDT , 'ORDER' RptType  from rm_cont a where  ISNULL(ARRIVALTYPE,'P')='P' group by  DIVCODE ,CONTDT 
  UNION ALL
select DISTINCT sum(a.ordKGS) ordqty , DIVCODE ,CONTDT , 'ORDER' RptType  from rm_cont a where ISNULL(ARRIVALTYPE,'P')='K' group by  DIVCODE ,CONTDT 
union all  
 select DISTINCT sum(ordqty-isnull(cancelbales,0)+isnull(recqty,0)) ordqty , DIVCODE ,CONTDT , 'Pending' RptType  from rm_cont a where  isnull(a.Fclosed,'N')<>'Y' and   (ordqty>isnull(cancelbales,0)+isnull(recqty,0))  and ISNULL(ARRIVALTYPE,'P')='P' group by  DIVCODE ,CONTDT 
  UNION ALL
select DISTINCT sum (ordKGS-isnull(rCDKGS,0)-ISNULL(a.cancelkgs ,0)) ordqty , DIVCODE ,CONTDT , 'Pending' RptType  from rm_cont a where  isnull(a.Fclosed,'N')<>'Y' and (ordKGS>(isnull(rCDKGS,0)-ISNULL(a.cancelkgs ,0)))  AND ISNULL(ARRIVALTYPE,'P')='K'  group by  DIVCODE ,CONTDT
 go

  from rm_lot a

on a.DIVCODE =d.DIVCODE and d.lotdt =a.LOTDT and d.LOTNO=a.LOTNO and a.varcode=d.varcode and d.BALENO=b.BALENO      
 --and d.LOTTYPE =a.LOTTYPE

 -- KSP_RMIDASHBORDMIXINGISSUE '01','2020-03-31','2020-03-31'
 create proc KSP_RMIDASHBORDMIXINGISSUE 
 (
 @divcode varchar(2),
 @Fdate varchar(15),
 @tdate varchar(15) 
 )as
 
 select c.MIXGRPNAME , round(isnull(dayIssue,0) /1000,3) onDateKg,  round(isnull(MONTHIssue,0)/1000,3) MONTHIssue ,round(isnull(yearIssue,0)/1000,3)  yearIssue 
  FROM RM_MIxGrp c

     left join (select a.DIVCODE ,b.MIXGRP ,SUM(ISSKGS ) dayIssue FROM RM_ISSB AS A 
  inner join RM_ISSH AS b on a.docno=b.docno and a.docdt=b.docdt and a.divcode=b.divcode 
  where a.docdt between  @Fdate and @tdate  group by a.DIVCODE ,b.MIXGRP 
   ) dd on dd.DIVCODE =c.DIVCODE and c.MIXGRPCD =dd.MIXGRP  
  left join (select a.DIVCODE ,b.MIXGRP ,SUM(ISSKGS ) MONTHIssue FROM RM_ISSB AS A 
  inner join RM_ISSH AS b on a.docno=b.docno and a.docdt=b.docdt and a.divcode=b.divcode 
  where a.docdt between  DATEADD(dd,-(DAY(@Fdate)),@Fdate) and  DATEADD(dd,-(DAY(DATEADD(mm,1,@Fdate))-1),
DATEADD(mm,1,@Fdate)) group by a.DIVCODE ,b.MIXGRP 
   ) d on d.DIVCODE =c.DIVCODE and c.MIXGRPCD =d.MIXGRP 
  left join (select a.DIVCODE ,b.MIXGRP ,SUM(ISSKGS ) yearIssue FROM RM_ISSB AS A 
  inner join RM_ISSH AS b on a.docno=b.docno and a.docdt=b.docdt and a.divcode=b.divcode 
  inner join PP_YEAR c on b.DOCDT between  c.AYFDATE and c.AYLDATE 
  where @Fdate between   c.AYFDATE and c.AYLDATE  group by a.DIVCODE ,b.MIXGRP 
   ) f on f.DIVCODE =c.DIVCODE and c.MIXGRPCD =f.MIXGRP
   where c.DIVCODE =@divcode 
  

go



 -- KSP_RMIDASHBORDVARISSUE '01','2020-03-31','2020-03-31'
 create proc KSP_RMIDASHBORDVARISSUE 
 (
 @divcode varchar(2),
 @Fdate varchar(15),
 @tdate varchar(15) 
 )as
 
 select c.varname ,ROUND( isnull(dayIssue,0)/1000,3) onDateKg, ROUND( isnull(MONTHIssue,0)/1000,3)MONTHIssue ,
 round(isnull(preMONTHIssue,0)/1000,3 ) preMONTHIssue ,round(isnull(yearIssue,0)/1000,2)  yearIssue
  FROM RM_var c

     left join (select a.DIVCODE ,a.VARCODE  ,SUM(ISSKGS ) dayIssue FROM RM_ISSB AS A 
  inner join RM_ISSH AS b on a.docno=b.docno and a.docdt=b.docdt and a.divcode=b.divcode 
  where a.docdt between  @Fdate and @tdate and a.DIVCODE =@divcode  group by a.DIVCODE ,a.VARCODE 
   ) dd on c.VARCODE   =dd.VARCODE  
  left join (select a.DIVCODE ,a.VARCODE ,SUM(ISSKGS ) MONTHIssue FROM RM_ISSB AS A 
  inner join RM_ISSH AS b on a.docno=b.docno and a.docdt=b.docdt and a.divcode=b.divcode 
  where a.docdt between  DATEADD(dd,-(DAY(@Fdate)),@Fdate) and  DATEADD(dd,-(DAY(DATEADD(mm,1,@Fdate))-1),
DATEADD(mm,1,@Fdate)) and a.divcode=@divcode group by a.DIVCODE ,a.VARCODE 
   ) d on c.VARCODE   =dd.VARCODE  
   
  inner join (select a.DIVCODE ,a.VARCODE  ,SUM(ISSKGS ) yearIssue FROM RM_ISSB AS A 
  inner join RM_ISSH AS b on a.docno=b.docno and a.docdt=b.docdt and a.divcode=b.divcode 
  inner join PP_YEAR c on b.DOCDT between  c.AYFDATE and c.AYLDATE 
  where @Fdate between   c.AYFDATE and c.AYLDATE  group by a.DIVCODE ,a.VARCODE 
   ) f on c.VARCODE   =f.VARCODE  
  left join (select a.DIVCODE ,a.VARCODE ,SUM(ISSKGS ) preMONTHIssue FROM RM_ISSB AS A 
  inner join RM_ISSH AS b on a.docno=b.docno and a.docdt=b.docdt and a.divcode=b.divcode 
  where a.docdt between DATEADD(dd,-(DAY(DATEADD(dd,-(DAY(@Fdate)),@Fdate))-1),DATEADD(dd,-(DAY(@Fdate )),@Fdate))
   and DATEADD(dd,-(DAY(@Fdate)),@Fdate) and a.divcode=@divcode group by a.DIVCODE ,a.VARCODE 
   ) ff on c.VARCODE   =ff.VARCODE  
 
  

 go




 -- KSP_RMIDASHBORD_VARORDERLIST '01','2020-03-31','2020-03-31'
 create proc KSP_RMIDASHBORD_VARORDERLIST
 (
 @divcode varchar(2),
 @Fdate varchar(15),
 @tdate varchar(15) 
 )as
 
 select c.varname ,ROUND( isnull(dayIssue,0)/1000,3) onDateKg, ROUND( isnull(MONTHIssue,0)/1000,3)MONTHORD ,
 round(isnull(preMONTHIssue,0)/1000,3 ) preMONTHORD ,round(isnull(yearIssue,0)/1000,2)  yearORD
  FROM RM_var c
     left join (select a.DIVCODE ,a.VARCODE  ,SUM(ordKGS ) dayIssue FROM RM_cont AS A 
  where a.contdt between  @Fdate and @tdate and a.DIVCODE =@divcode  group by a.DIVCODE ,a.VARCODE 
   ) dd on c.VARCODE   =dd.VARCODE  
  left join (select a.DIVCODE ,a.VARCODE ,SUM(ordKGS ) MONTHIssue FROM RM_cont AS A 

  where a.CONTDT  between  DATEADD(dd,-(DAY(@Fdate)),@Fdate) and  DATEADD(dd,-(DAY(DATEADD(mm,1,@Fdate))-1),
DATEADD(mm,1,@Fdate)) and a.divcode=@divcode group by a.DIVCODE ,a.VARCODE 
   ) d on c.VARCODE   =dd.VARCODE  
   
  inner join (select a.DIVCODE ,a.VARCODE  ,SUM(ordKGS ) yearIssue FROM RM_cont AS A 
  
  inner join PP_YEAR c on a.CONTDT between  c.AYFDATE and c.AYLDATE 
  where @Fdate between   c.AYFDATE and c.AYLDATE  group by a.DIVCODE ,a.VARCODE 
   ) f on c.VARCODE   =f.VARCODE  
  left join (select a.DIVCODE ,a.VARCODE ,SUM(ordKGS ) preMONTHIssue FROM RM_cont AS A 

  where a.CONTDT  between DATEADD(dd,-(DAY(DATEADD(dd,-(DAY(@Fdate)),@Fdate))-1),DATEADD(dd,-(DAY(@Fdate )),@Fdate))
   and DATEADD(dd,-(DAY(@Fdate)),@Fdate) and a.divcode=@divcode group by a.DIVCODE ,a.VARCODE 
   ) ff on c.VARCODE   =ff.VARCODE  
 
  

go

create view VW_RMI_DB_PendingORDSUP
as
select DISTINCT round(sum (ordKGS-isnull(c.netwt ,0)-(ISNULL(a.cancelkgs ,0) +  ((ORDKGS /ORDQTY)*isnull(a.CancelBales ,0) ) ))/1000,3) ordqty , a.DIVCODE 
,a.CONTDT ,b.slname,a.CONTNO ,b.slcode ,f.VARNAME,a.varcode   from rm_cont a 
inner join RM_VAR f on f.VARCODE =a.VARCODE 
inner join fa_slmas b on b.slcode =a.SUPCD 
left join (select DIVCODE ,CONTNO ,CONTDT ,VARCODE ,CATCD ,SUM(NETWT ) netwt from RM_ARRIVAL 
group by DIVCODE ,CONTNO ,CONTDT ,VARCODE ,CATCD 
) c on c.VARCODE  =a.VARCODE and  a.CONTNO =c.CONTNO and a.CONTDT =c.CONTDT and c.DIVCODE =a.DIVCODE 

where  isnull(a.Fclosed,'N')<>'Y' and round((ordKGS -isnull(c.netwt ,0)-((ISNULL(a.cancelkgs ,0) +  ((ORDKGS /ORDQTY)*isnull(a.CancelBales ,0) ) )))/1000,3) >0
 group by  a.DIVCODE ,a.CONTDT,b.slname ,a.CONTNO ,b.slcode ,f.VARNAME,a.varcode


go




 -- KSP_RMIDASHBORD_VARRECIPTLIST '01','2020-03-31','2020-03-31'
 create proc KSP_RMIDASHBORD_VARRECIPTLIST
 (
 @divcode varchar(2),
 @Fdate varchar(15),
 @tdate varchar(15) 
 )as
 
 select c.varname ,ROUND( isnull(dayIssue,0)/1000,3) onDateKg, ROUND( isnull(MONTHIssue,0)/1000,3)MONTHRec ,
 round(isnull(preMONTHIssue,0)/1000,3 ) preMONTHREC ,round(isnull(yearIssue,0)/1000,2)  yearREC, ROUND(DayVal/100000,2) DayVal
 , ROUND(Monthval/100000,2) Monthval,ROUND(YearVal/100000,2) YearVal,ROUND(PrvMonthval/100000,2) PrvMonthval
 
  FROM RM_var c
     left join (select a.DIVCODE ,a.VARCODE  ,SUM(favaourablewgt  ) dayIssue,SUM(a.favaourablewgt *RATEKG)  DayVal FROM RM_lot AS A 
  where a.lotdt between  @Fdate and @tdate and a.DIVCODE =@divcode    and ISNULL(OPFLG ,'N')<>'Y' group by a.DIVCODE ,a.VARCODE 
   ) dd on c.VARCODE   =dd.VARCODE  
  left join (select a.DIVCODE ,a.VARCODE ,SUM(favaourablewgt ) MONTHIssue,SUM(a.favaourablewgt *RATEKG) Monthval FROM RM_lot AS A 

  where a.lotdt  between  DATEADD(dd,-(DAY(@Fdate)),@Fdate) and  DATEADD(dd,-(DAY(DATEADD(mm,1,@Fdate))-1),
DATEADD(mm,1,@Fdate)) and a.divcode=@divcode 
   and ISNULL(OPFLG ,'N')<>'Y'
group by a.DIVCODE ,a.VARCODE 
   ) d on c.VARCODE   =dd.VARCODE  
   
  inner join (select a.DIVCODE ,a.VARCODE  ,SUM(favaourablewgt ) yearIssue,SUM(a.favaourablewgt *RATEKG) YearVal FROM RM_lot AS A 
  
  inner join PP_YEAR c on a.LOTDT between  c.AYFDATE and c.AYLDATE 
  where @Fdate between   c.AYFDATE and c.AYLDATE 
     and ISNULL(OPFLG ,'N')<>'Y'
   group by a.DIVCODE ,a.VARCODE 
   ) f on c.VARCODE   =f.VARCODE  
  left join (select a.DIVCODE ,a.VARCODE ,SUM(favaourablewgt ) preMONTHIssue,SUM(a.favaourablewgt *RATEKG) PrvMonthval FROM RM_lot AS A 

  where a.lotdt  between DATEADD(dd,-(DAY(DATEADD(dd,-(DAY(@Fdate)),@Fdate))-1),DATEADD(dd,-(DAY(@Fdate )),@Fdate))
   and DATEADD(dd,-(DAY(@Fdate)),@Fdate) and a.divcode=@divcode 
   and ISNULL(OPFLG ,'N')<>'Y'
   group by a.DIVCODE ,a.VARCODE 
   ) ff on c.VARCODE   =ff.VARCODE  


go
