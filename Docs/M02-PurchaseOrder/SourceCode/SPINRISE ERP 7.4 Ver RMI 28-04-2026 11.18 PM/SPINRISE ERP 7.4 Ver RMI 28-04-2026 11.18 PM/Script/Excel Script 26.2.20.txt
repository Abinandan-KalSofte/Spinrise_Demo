alter PROC KSP_RMI_P_IssueList_Date                           
(                                    
 @Divcode VARCHAR(2),                                    
 @FDate VARCHAR(10),                                    
 @TDate VARCHAR(10),                                
 @IssueType VARCHAR(10),                                
 @LotType VARCHAR(10)                                
)                                    
AS                                    
BEGIN                                    
SELECT x.DOCDT,x.DOCNO,X.cntcd,X.cntname,X.VARCODE,X.VARNAME,X.LOTNO,X.LOTYEAR,                                  
SUM(ISNULL(BALES,0))AS BALES,SUM(ISNULL(NETWT,0))AS NETWT,                                    
SUM(ISNULL(INVWT,0))AS INVWT,X.[Rate/Qtl],(SUM(ISNULL(NETWT,0))/100)*[Rate/Qtl] TOTLANDCOST,X.RATECY,X.DIV_PRINTNAME,X.DIV_UNITNAME,                                  
--CONVERT(VARCHAR,X.LOTNO)+'/'+RIGHT(CONVERT(VARCHAR,((X.LOTYEAR))),2)+'-'+RIGHT(CONVERT(VARCHAR,(X.LOTYEAR)+1),2)LotNO_FYear,                                  
X.CATCD,X.CATNAME,CONVERT(VARCHAR,X.LOTNO)+'/'+right(X.AYEAR,5) AS LotNO_FYear  ,SUM(ISNULL(X.ISSKGS,0))  AS ISSKGS 
                             
FROM                                    
(select distinct a.docno, a.docdt,F.cntcd,G.cntname,A.VARCODE, b.varname,A.LOTNO,C.LOTYEAR,                                  
count(a.baleno) as bales,0 as netwt,0 as invwt,                                    
C.RATEKG*100 AS 'Rate/Qtl',C.TOTLANDCOST,C.RATECY AS RATECY,D.DIV_PRINTNAME,D.DIV_UNITNAME,                                  
C.CATCD,H.CATNAME,py.AYEAR ,0 as ISSKGS                                  
from rm_issb a                                    
INNER JOIN RM_ISSH F ON F.DIVCODE = a.DIVCODE AND F.DOCNO = a.DOCNO AND F.DOCDT = a.DOCDT AND F.ISSTYPE = a.ISSTYPE                                    
inner JOIN rm_var b ON b.VARCODE = a.VARCODE                                    
left JOIN RM_LOT C ON C.DIVCODE=A.DIVCODE AND C.LOTNO=A.LOTNO AND C.LOTDT=A.lotdt AND C.LOTTYPE=A.LOTTYPE AND                                     
C.VARCODE=A.VARCODE   and c.catcd = a.catcd                              
LEFT OUTER JOIN PP_YEAR py on C.LOTDT BETWEEN py.AYFDATE AND py.AYLDATE                              
LEFT JOIN PP_DIVMAS D ON D.DIVCODE=A.DIVCODE                                    
LEFT JOIN RM_COUNT G ON G.Divcode=F.DIVCODE AND G.cntcd=F.cntcd AND G.mixgrpcd=F.MIXGRP                                    
LEFT JOIN RM_CAT H ON H.CATCD=C.CATCD                                  
where     
--a.issued='Y'  and    
 a.Divcode=@Divcode and a.docdt between @FDate and  @TDate AND                              (@IssueType='A' or a.ISSTYPE =@IssueType) and         
(@LotType ='ALL' or a.LOTTYPE= @LotType)                
--((@IssueType='A' AND a.ISSTYPE IN (SELECT  distinct isstype FROM  rm_issb))  OR                                  
--(@IssueType<> 'A' AND a.ISSTYPE IN (SELECT  distinct isstype FROM  rm_issb WHERE isstype =@IssueType )))AND                                
--((@LotType ='ALL' AND a.LOTTYPE IN (SELECT distinct LOTTYPE FROM  rm_issb))  OR                                  
--(@LotType <> 'ALL'  AND a.LOTTYPE IN (SELECT distinct  LOTTYPE FROM  rm_issb WHERE LOTTYPE =@LotType  )))                                 
AND C.LOTYEAR=year(py.ayfdate)                     
--AND C.LOTYEAR=year(a.docdt)                     
                         
group by a.docno, a.docdt,F.cntcd,G.cntname,A.VARCODE,b.varname,A.LOTNO,C.LOTYEAR,C.RATEKG,C.TOTLANDCOST,C.RATECY,                                    
D.DIV_PRINTNAME,D.DIV_UNITNAME,C.CATCD,H.CATNAME,py.AYEAR                               
                           
Union All                                    
                            
select a.docno, a.docdt,F.cntcd,G.cntname, A.VARCODE,b.varname, A.LOTNO,C.LOTYEAR,                                  
0 as BALES, sum(ISNULL(a.actisskgs,0))as netwt,                                    
sum(ISNULL(c.pnetwt,0))as invwt,C.RATEKG*100 AS 'Rate/Qtl',C.TOTLANDCOST,C.RATECY AS RATECY,                                    
E.DIV_PRINTNAME,E.DIV_UNITNAME,                               
C.CATCD,H.CATNAME,py.AYEAR,sum(a.ISSKGS) as ISSKGS            
from rm_issb a                                    
INNER JOIN RM_ISSH F ON F.DIVCODE = a.DIVCODE AND F.DOCNO = a.DOCNO AND F.DOCDT = a.DOCDT AND F.ISSTYPE = a.ISSTYPE                                    
inner JOIN PP_YEAR py1 on a.docdt BETWEEN py1.AYFDATE AND py1.AYLDATE               
left JOIN rm_var b ON b.VARCODE = a.VARCODE                                    
inner JOIN RM_LOT C ON C.DIVCODE=A.DIVCODE AND C.LOTNO=A.LOTNO AND C.LOTDT=A.lotdt AND C.LOTTYPE=A.LOTTYPE AND                                     
C.VARCODE=A.VARCODE     and c.catcd = a.catcd                            
inner  JOIN PP_YEAR py on C.LOTDT BETWEEN py.AYFDATE AND py.AYLDATE                                 
left JOIN rm_bale D ON D.DIVCODE=A.DIVCODE AND D.LOTNO=A.LOTNO AND D.lotdt=A.lotdt AND D.lottype=A.LOTTYPE AND D.LOTNO=c.LOTNO AND D.lotdt=c.lotdt and d.VARCODE =c.VARCODE  AND       C.LOTTYPE=d.lottype     AND                          
D.BALENO=A.BALENO    and d.VARCODE =a.VARCODE and d.CATCD = a.CATCD                           
inner JOIN PP_DIVMAS E ON E.DIVCODE=A.DIVCODE                                    
left JOIN RM_COUNT G ON G.Divcode=F.DIVCODE AND G.cntcd=F.cntcd AND G.mixgrpcd=F.MIXGRP                                    
LEFT JOIN RM_CAT H ON H.CATCD=C.CATCD                                  
where     
--a.issued='Y' and     
a.Divcode=@Divcode and a.docdt between @FDate and @TDate AND                         (@IssueType='A' or a.ISSTYPE =@IssueType) and         
(@LotType ='ALL' or a.LOTTYPE= @LotType)         
--((@IssueType='A' AND a.ISSTYPE IN (SELECT distinct  isstype FROM  rm_issb))  OR                                  
--(@IssueType<> 'A' AND a.ISSTYPE IN (SELECT distinct isstype FROM  rm_issb WHERE isstype =@IssueType )))AND                                
--((@LotType ='ALL' AND a.LOTTYPE IN (SELECT distinct LOTTYPE FROM  rm_issb))  OR                                  
--(@LotType <> 'ALL'  AND a.LOTTYPE IN (SELECT distinct  LOTTYPE FROM  rm_issb WHERE LOTTYPE =@LotType  )))                                
--AND C.LOTYEAR=year(py1.ayfdate)                     
AND C.LOTYEAR=year(py.ayfdate)                  
--AND a.lotno =41                        
group by a.docno, a.docdt,F.cntcd,G.cntname,A.VARCODE,b.varname,A.LOTNO,C.LOTYEAR,C.RATEKG,C.TOTLANDCOST,C.RATECY,                                    
E.DIV_PRINTNAME,E.DIV_UNITNAME,C.CATCD,H.CATNAME,py.AYEAR        
        
        
)X                                    
GROUP BY x.DOCNO,x.DOCDT,X.cntcd,X.cntname,X.VARCODE,X.VARNAME,X.LOTNO,X.LOTYEAR,X.[Rate/Qtl],X.TOTLANDCOST,X.RATECY,                                    
X.DIV_PRINTNAME,X.DIV_UNITNAME,X.CATCD,X.CATNAME,X.AYEAR--,X.ISSKGS                                  
ORDER BY X.docdt,X.docno,X.CNTCD,X.VARCODE                                 
                            
END

alter Procedure ksp_mixnumber_wisenew                                                                     
@divcode varchar(2),                                                                                            
@FromDate AS varchar(15),                                                                                               
@ToDate AS Varchar(15),                  
@lotyear AS Varchar(10)                  
                                                                                     
As                        
begin                      
                      
select distinct                     
docno,    
--mixgrpcd as mixgrp,                    
cntcode as mixgrp,  
docdt,varcode,catcd,cntname ,catname,lotno,sum(bale) as bale,                      
unit ,round(sum(isnull(netkgs,0)),3)netkg                    
,sum(isnull(quintal,0)) ratequin,                      
sum(isnull(Amount,0)) AS amount,avg(ratekg) as avgrate,plotno,                    
round(avg(ratekg*355.616),0)as ratecandy,                      
mixgrpcd,cntcode,DIV_PRINTNAME,DIV_UNITNAME   from (                    
SELECT c.DOCDT, a.CNTCODE AS CNTCD, d.mixgrpcd, a.VARCODE, a.LOTNO, COUNT(b.BALENO) AS bale,                     
(CASE WHEN a.bblflg = 'R' THEN 'Borah' ELSE 'Bales' END) AS Unit, SUM(ISNULL(b.ISSKGS, 0)) AS 'netkgs', a.PLOTNO, AVG(a.RATEKG) AS 'quintal',                     
SUM(ISNULL(b.ISSKGS, 0)) * a.RATEKG AS AMOUNT, a.RATEKG, ROUND(a.RATEKG * 355.616, 0) AS ratecandy, ISNULL(c.DOCNO, 0) AS docno,                     
a.catcd,f.CATNAME ,p.DIV_PRINTNAME,p.DIV_UNITNAME, c.cntcd AS cntcode  ,d.cntname                  
FROM dbo.RM_ISSB AS b   
INNER JOIN                    
dbo.RM_ISSH AS c ON b.DIVCODE = c.DIVCODE AND b.ISSTYPE = c.ISSTYPE AND b.DOCDT = c.DOCDT AND b.DOCNO = c.DOCNO    
left JOIN                    
 dbo.RM_BALE AS e ON e.DIVCODE = b.DIVCODE AND e.lottype = b.LOTTYPE AND e.LOTNO = b.LOTNO AND e.CATCD = b.CATCD AND                     
e.BALENO = b.BALENO left JOIN                    
dbo.RM_LOT AS a ON e.lotdt = a.LOTDT AND b.DIVCODE = a.DIVCODE AND b.CATCD = a.CATCD AND b.LOTNO = a.LOTNO AND                     
b.lotdt = a.LOTDT   
                   
left JOIN dbo.RM_CAT as f ON b.CATCD=f.CATCD         
inner join PP_DIVMAS p on p.DIVCODE=b.DIVCODE                     
LEFT OUTER JOIN dbo.RM_COUNT AS d ON c.cntcd = d.cntcd  and c.divcode=d.divcode                  
WHERE (b.LOTTYPE in ('A','T')) AND (a.DIVCODE = @divcode) AND (a.REJFLG = 'N') AND (b.issued = 'N' OR b.issued = 'Y')           
AND (a.LOTYEAR = @lotyear)                    
and c.docdt  between @Fromdate  and @Todate                       
GROUP BY d.mixgrpcd, a.BBLFLG, a.RATEKG, a.RATECY, a.VARCODE, a.LOTNO, c.DOCDT, a.CNTCODE, a.PLOTNO, c.DOCNO, a.CATCD,f.CATNAME, c.cntcd,DIV_PRINTNAME,DIV_UNITNAME ,d.cntname                    
) xxx                        
group by varcode,lotno,DOCDT,unit,plotno,                    
docno,                    
catcd,catname,mixgrpcd,cntcode,DIV_PRINTNAME,DIV_UNITNAME,cntname,CNTCD                    
order by cntcode                      
                      
END
