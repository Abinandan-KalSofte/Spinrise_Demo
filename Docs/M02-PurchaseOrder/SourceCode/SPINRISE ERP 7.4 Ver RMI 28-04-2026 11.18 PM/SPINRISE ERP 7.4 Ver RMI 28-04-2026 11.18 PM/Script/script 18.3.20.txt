create view VW_RMI_FibreCreditNoteDiscount
as
select  a.docdate ,d.VARname 'Fibre', c.slname 'Supplier',e.PLName 'PlantName',f.Description 'Disctype',
g.comwt,cast(round(g.comwt*b.drate,2) as numeric(15,2)) DiscValue,b.drate  , a.fdate , a.tdate,a.Plant,a.varcode,a.supcd  from  RM_FibreDiscCN  a
inner join RM_FibreDiscCNDRate  b on a.DOCNO=b.docno and a.docdate=b.docdate and a.supcd=b.supcd and a.varcode=b.varcode
inner join fa_slmas c on a.supcd=c.slcode 
inner join rm_var d on d.varcode=a.varcode 
inner join rm_plant e on e.PLCode=a.Plant 
inner join RM_DiscountType f on f.Code =b.DType and RPTFLG='Y'
left join (select  lotdt,sum(comm_wt) comwt,supcd,varcode  from rm_lot group by supcd,varcode,lotdt ) g
on g.varcode=a.varcode and g.supcd=a.supcd and g.lotdt between a.fdate and a.tdate

create view Vw_RM_CreditNoteDiscountTypes 
as
select  Description  dicsctype from RM_DiscountType where RPTFLG='Y'
