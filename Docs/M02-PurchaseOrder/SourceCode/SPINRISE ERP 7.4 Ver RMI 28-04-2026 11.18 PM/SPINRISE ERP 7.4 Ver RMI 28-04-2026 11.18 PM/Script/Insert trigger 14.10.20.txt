          narration1,credit,trntype)            
       VALUES(@v_divcode,@v_insgl,@v_invno,@v_invdt,2,@v_invtc,            
          --'PURCHASE RETURN'  
   @v_narr1,ISNULL(@v_insamt,0),'PR')            
    ELSE            
          UPDATE fa_cgtrn            
         SET credit = ISNULL(credit, 0) + ISNULL(@v_insamt, 0)            
          WHERE divcode = @v_divcode AND glcode = @v_insgl AND            
          vocdt = @v_invdt AND tc = @v_invtc AND vocno = @v_invno AND            
          vocsno = 2 and trntype = 'PR'             
   END            
               
-- ROUND-OFF            
            
   IF (NOT ISNULL(@v_roff,0) =0 ) AND (@v_roffgl IS NOT NULL AND @v_roffgl <> '')             
   BEGIN            
            
      IF @v_roff >0             
        BEGIN            
     SET @v_cnt = 0            
  SELECT @v_cnt = COUNT(*) from fa_cgtrn            
     WHERE divcode = @v_divcode AND glcode = @v_roffgl AND            
   vocdt = @v_invdt AND tc = @v_invtc AND vocno = @v_invno AND            
           vocsno = 2 and trntype = 'PR'            
  if @v_cnt=0            
           INSERT INTO fa_cgtrn (divcode,glcode,vocno,vocdt,vocsno,tc,            
              narration1,credit,trntype)            
           VALUES(@v_divcode,@v_roffgl,@v_invno,@v_invdt,2,@v_invtc,            
--              'PURCHASE RETURN'  
  @v_narr1,ISNULL(@v_roff,0),'PR')            
  ELSE            
   UPDATE fa_cgtrn            
           SET credit = ISNULL(credit, 0) + ISNULL(@v_roff, 0)            
           WHERE divcode = @v_divcode AND glcode = @v_roffgl AND            
           vocdt = @v_invdt AND tc = @v_invtc AND vocno = @v_invno AND            
           vocsno = 2 and trntype = 'PR'             
             END                  
      ELSE            
       BEGIN              
     SET @v_cnt = 0            
  SELECT @v_cnt = COUNT(*) from fa_cgtrn            
     WHERE divcode = @v_divcode AND glcode = @v_roffgl AND          
           vocdt = @v_invdt AND tc = @v_invtc AND vocno = @v_invno AND            
           vocsno = 1 and trntype = 'PR'            
  IF @v_cnt=0            
            INSERT INTO fa_cgtrn (divcode,glcode,vocno,vocdt,vocsno,tc,            
               narration1,debit,trntype)            
            VALUES(@v_divcode,@v_roffgl,@v_invno,@v_invdt,1,@v_invtc,            
            --   'PURCHASE RETURN'  
  @v_narr1,ABS(ISNULL(@v_roff,0)),'PR')            
    ELSE            
   UPDATE fa_cgtrn            
            SET debit = ISNULL(debit, 0) + ABS(ISNULL(@v_roff, 0))            
            WHERE divcode = @v_divcode AND glcode = @v_roffgl AND            
            vocdt = @v_invdt AND tc = @v_invtc AND vocno = @v_invno AND            
            vocsno = 1 and trntype = 'PR'             
                   END            
 END            
            
--INVOICE AMOUNT            
            
             
            
   BEGIN            
       INSERT INTO fa_cstrn (divcode,slcode,vocdt,tc,vocno,vocsno,            
        narration1,debit,trntype)            
       VALUES(@v_divcode,@v_slcode,@v_invdt,@v_invtc,@v_invno,1,            
        --  'PURCHASE RETURN'  
  @v_narr1,ISNULL(@v_invamt,0),'PR')            
             
     -- CONTROL A/C            
                  
       set @v_cnt = 0            
                  
       SELECT @v_cglcode = glcode  FROM fa_prty            
       WHERE prtytyp = substring(@v_slcode,1,2)            
            
       SELECT @v_cnt = COUNT(*)  FROM fa_cgtrn            
       WHERE divcode = @v_divcode AND glcode = @v_cglcode AND            
       vocdt = @v_ldate AND tc = @v_abstc AND vocno = @v_gvocno AND            
       vocsno = 1 AND trntype = 'AB'            
            
       IF @v_cnt > 0             
          UPDATE fa_cgtrn            
          SET debit = ISNULL(debit, 0) + ISNULL(@v_invamt, 0)            
          WHERE divcode = @v_divcode AND glcode = @v_cglcode AND            
             vocdt = @v_ldate AND tc = @v_abstc AND vocno = @v_gvocno AND            
             vocsno = 1 AND trntype = 'AB'            
              ELSE            
            
                INSERT INTO fa_cgtrn            
          (divcode,glcode,tc,vocno,vocdt,vocsno,trntype,narration1,debit)            
VALUES(@v_divcode,@v_cglcode,@v_abstc,@v_gvocno,@v_ldate,1,'AB',            
          'Sales Summary',ISNULL(@v_invamt,0))            
            
  -- UPDATION OF ACCOUNTS RECEIVABLES TABLE            
             
       IF substring(@v_slcode,1,1) = 'C'             
                    INSERT INTO fa_apdbt            
              (divcode, slcode, trntype, tc, vocdt, vocno, vocsno, narr1,             
              debit, balance)            
          VALUES(@v_divcode, @v_slcode, 'PR', @v_invtc, @v_invdt, @v_invno, 1,            
        --      'PURCHASE RETURN'  
   @v_narr1, @v_invamt, @v_invamt)            
       ELSE            
           IF substring(@v_slcode,1,1) = 'D'             
                        INSERT INTO fa_arecm            
                  (divcode, slcode,agent_code, trntype, invtc, invdt, invno, invsno,            
                  ordno, orddt, narr1, crdays, paymode, invamt, amtdue)            
               VALUES(@v_divcode, @v_slcode,isnull(@v_agentcode,''),'PR', @v_invtc, @v_invdt, @v_invno,            
                  1,@v_ordno, @v_orddt, @v_narr1, @v_crdays, @v_paymode,            
                  @v_invamt, @v_invamt)           
   --'PURCHASE RETURN'   
    END            
            
 END            
end
