--  ksp_CottonCost '2017-04-01','2018-08-01','01','2017'          
          
create PROCEDURE ksp_CottonCost          
@fdate AS varchar(10),          
@tdate AS varchar(10),          
@divcode AS varchar(10),          
@lotyear AS varchar(10)           
AS           
          
select a.lotdt MillLotDt,a.arrdt ReceiptDt,z.varname as varcode ,a.lotno MillLotno,  
a.PLOTNO partylotno,  
bales= (case when BBLFLG ='B' then bales ELSE 0 end)+(case when bblflg<>'B' then bales ELSE 0 end)    
, round(a.netwt,2) as Qtyinkgs, a.RATECY RateperCandy,a.RATEKG ,    
ISNULL(a.pjamt,0) AS InvoiceAmt,  0.00 premimumAmt,    
ISNULL(a.FREGHT ,0)  FREGHT,ISNULL(a.COMMAMT,0) COMMAMT, isnull(a.totlandcost,0) totlandcost  ,  
cc.dlytype,f.slname,s.div_printname,s.div_unitname ,mg.MIXGRPNAME 
     
    
    
from  rm_lot a     

inner join rm_mixchartdt aa on aa.lotno =a.lotno and aa.lotdt=a.lotdt and a.divcode=aa.divcode and a.varcode=aa.varcode 
inner join rm_mixcharthd m on aa.docno=m.docno and aa.docdt=m.docdt and aa.divcode=m.divcode
inner join RM_MIXGRP mg on m.mixgrp=mg.MIXGRPCD
inner join rm_arrival bb on a.arrno=bb.arrno and bb.arrdate=a.arrdt and a.supcd=bb.supcd and a.divcode=bb.divcode and a.varcode=bb.varcode  
inner join rm_cont cc on cc.contno=bb.contno and bb.contdt=cc.contdt and a.divcode=cc.divcode and a.supcd=cc.supcd and a.varcode=cc.varcode    
inner join rm_var z on a.varcode=z.varcode     
inner join  fa_slmas f  on  a.supcd=f.slcode     
inner join   pp_divmas s  on   a.divcode=s.divcode          
          
where a.divcode=@divcode and a.opflg in('N','n') and  a.lotyear=@lotyear and           
a.lotdt BETWEEN @fdate AND @tdate        
-- and pjdt BETWEEN @fdate AND @tdate        
            
--and pjno is not null        
           
          
ORDER BY a.LOTDT , a.arrdt, a.varcode
