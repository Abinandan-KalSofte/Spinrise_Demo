--SP_CONSRMIDENIER


ALTER PROC SP_CONSRMIDENIER
AS 
select CATNAME,GROUPNAME,VARCODE,VARNAME,UNIT,VARDESC,SHADENAME,SHADENO,DENIER,LENGTH,supplier,
sum(PTL_I_Qty)PTL_I_Qty,Sum(PTL_I_NetKgs)PTL_I_NetKgs,Sum(PTL_I_Value)PTL_I_Value,
sum(PTL_II_Qty)PTL_II_Qty,Sum(PTL_II_NetKgs)PTL_II_NetKgs,Sum(PTL_II_Value)PTL_II_Value,
sum(SCSL_I_Qty)SCSL_I_Qty,Sum(SCSL_I_NetKgs)SCSL_I_NetKgs,Sum(SCSL_I_Value)SCSL_I_Value,
sum(SCSL_II_Qty)SCSL_II_Qty,Sum(SCSL_II_NetKgs)SCSL_II_NetKgs,Sum(SCSL_II_Value)SCSL_II_Value,
sum(isnull(PTL_I_Qty,0))+sum(ISNULL(PTL_II_Qty,0))+sum(ISNULL(SCSL_I_Qty,0))+sum(ISNULL(SCSL_II_Qty,0)) AS TOTQTY,
Sum(ISNULL(PTL_I_NetKgs,0))+Sum(ISNULL(PTL_II_NetKgs,0))+Sum(ISNULL(SCSL_I_NetKgs,0))+Sum(ISNULL(SCSL_II_NetKgs,0)) AS TOTKGS,
Sum(ISNULL(PTL_I_Value,0))+Sum(ISNULL(PTL_II_Value,0))+Sum(ISNULL(SCSL_I_Value,0))+Sum(ISNULL(SCSL_II_Value,0)) AS TOTVAL


FROM
(
SELECT abbr,CATNAME,VARCODE,VARNAME,UNIT,VARDESC,SHADENAME,SHADENO,DENIER,LENGTH,LOTNO,PLOTNO,SuPPLIER,
Sum(CASE WHEN ABBR='PTL-I' THEN BALESBORASQTY end) AS PTL_I_Qty,
Sum(CASE WHEN ABBR='PTL-I' THEN STOCK end) AS PTL_I_NetKgs,
Sum(CASE WHEN ABBR='PTL-I' THEN VAL end) AS PTL_I_Value,

Sum(CASE WHEN ABBR='PTL-II' THEN BALESBORASQTY end) AS PTL_II_Qty,
Sum(CASE WHEN ABBR='PTL-II' THEN STOCK end) AS PTL_II_NetKgs,
Sum(CASE WHEN ABBR='PTL-II' THEN VAL end) AS PTL_II_Value,

Sum(CASE WHEN ABBR='SCSL-I' THEN BALESBORASQTY end) AS SCSL_I_Qty,
Sum(CASE WHEN ABBR='SCSL-I' THEN STOCK end) AS SCSL_I_NetKgs,
Sum(CASE WHEN ABBR='SCSL-I' THEN VAL end) AS SCSL_I_Value,

Sum(CASE WHEN ABBR='SCSL-II' THEN BALESBORASQTY end) AS SCSL_II_Qty,
Sum(CASE WHEN ABBR='SCSL-II' THEN STOCK end) AS SCSL_II_NetKgs,
Sum(CASE WHEN ABBR='SCSL-II' THEN VAL end) AS SCSL_II_Value,GROUPNAME
from CONSRMIDENIER
group by abbr,catname,GROUPNAME,VARCODE,VARNAME,UNIT,VARDESC,SHADENAME,SHADENO,DENIER,LENGTH,LOTNO,PLOTNO,SuPPLIER)x
GROUP BY 
CATNAME,GROUPNAME,VARCODE,VARNAME,UNIT,VARDESC,SHADENAME,SHADENO,DENIER,LENGTH,supplier
