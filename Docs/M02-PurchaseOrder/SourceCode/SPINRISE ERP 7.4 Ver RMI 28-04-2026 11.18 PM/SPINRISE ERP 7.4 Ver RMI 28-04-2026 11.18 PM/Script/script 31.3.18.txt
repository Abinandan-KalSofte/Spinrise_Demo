--ksp_PO_RMBillMonitor '01','2018-04-01','2018-05-30'    
create PROC ksp_PO_RMBillMonitor        
(        
 @Divcode VARCHAR(2),        
 @FDate VARCHAR(10),        
 @TDate VARCHAR(10)        
    
)        
AS        
BEGIN     


select  distinct a.prdate,a.prno,b.varname itemname,case when a.BBFLAG='B' then 'Bale' else '' end uom,
a.ORDKGS qtyind,a.APPDATE directappdate,a.CONTNO pordno,a.CONTDT porddt,c.ARRDATE  docdt,        
 c.ARRNO  docno,d.PNETWT quantity,d.PJNO  pjvocno,d.PJDT pjvocdt,f.vocdt,f.vocno,f.lpayvocno,f.lpayvocdt,p.DIVNAME,p.ADD1,p.DIV_PRINTNAME,p.DIV_UNITNAME,DATEDIFF(d,a.prdate,a.APPDATE) AS PR_APPDays,        
DATEDIFF(d,a.prdate,c.CONTDT)AS PR_PODays,DATEDIFF(d,a.CONTDT,d.ARRDT)AS PR_MRNDays,
DATEDIFF(d,a.CONTDT ,d.ARRDT)AS PO_MRNDays,        
DATEDIFF(d,d.ARRDT,d.PJDT)AS MRN_PJVDays,DATEDIFF(d,d.ARRDT,f.LPAYVOCDT)AS MRN_PayDays,
DATEDIFF(d,d.PJDT ,f.LPAYVOCDT)AS PJV_PayDays  ,'' depcode,'' depname      
from  rm_cont a
inner join rm_var b on a.VARCODE=b.VARCODE 
left join rm_arrival  c on a.CONTNO=c.CONTNO and c.CONTDT =a.CONTDT and a.DIVCODE =c.DIVCODE and a.SUPCD =c.SUPCD and a.VARCODE=c.VARCODE
left join rm_lot d on d.DIVCODE=a.DIVCODE and a.SUPCD =d.SUPCD and d.VARCODE=a.VARCODE and c.ARRDATE =d.ARRDT and d.ARRNO =c.ARRNO 
left join fa_apaym f  on d.PJTC = f.tc and d.PJNO = f.vocno and d.PJDT  =f.vocdt and f.DIVCODE=d.DIVCODE
inner join PP_DIVMAS p  on a.DIVCODE=a.DIVCODE      
where        
a.divcode=@Divcode AND  a.contdt BETWEEN @FDate AND @TDate --AND a.Depcode BETWEEN @FDep AND @TDep        
END
