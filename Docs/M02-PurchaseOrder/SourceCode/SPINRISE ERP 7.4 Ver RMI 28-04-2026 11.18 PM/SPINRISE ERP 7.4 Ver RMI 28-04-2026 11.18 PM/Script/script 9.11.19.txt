create Proc KSP_OrderAmentment
(
--KSP_OrderAmentment '02','2019-04-01','2019-11-11',''
@divcode varchar(2),
@Fdate varchar(15),
@Tdate varchar(15),
@IPadd varchar(18)
)
as


select distinct  a.AContNo AmdPONo,a.ACONTDT AmdPOdt, a.contdt,c.varname,b.slname as SupplierName,
z.slname as BrokerName,v.areaname,
cast(a.contno as varchar) as contno,round(bb.candyrate,0) AS CandyRate,
unit=case when a.dlytype='F' then 'FOR' else 'SPOT' end,
isnull(a.ordqty,0)ordqty,a.duedate,
isnull(a.ordkgs,0)ordkgs,a.MillRefNo,a.PRVRatekg,a.PRVORDQty,a.PRVORDkgs,pp.DIV_PRINTNAME ,pp.DIV_UNITNAME 
,round(a.candyrate,0)*isnull(a.ordkgs,0) Amdvalue,round(isnull(a.PRVRatekg,0),0)*isnull(isnull(bb.PRVORDkgs,0),0) PrvOrdValue
from rm_acont a 
inner join RM_CONT bb on a.CONTNO=bb.contno and bb.CONTDT=a.CONTDT and a.DIVCODE =bb.DIVCODE and a.VARCODE =bb.VARCODE 
inner Join fa_slmas b on a.SUPCD=b.SLCODE 
Left Join  fa_slmas z on a.brkcd =z.slcode  
left JOin rm_var c on c.varcode=a.varcode 
Left Join rm_area v on a.areacOdE=v.areacode 
inner join pp_divmas pp on pp.DIVCODE=a.DIVCODE 
where a.acontdt Between @Fdate and @Tdate
 and a.divcode=@divcode
 
go
