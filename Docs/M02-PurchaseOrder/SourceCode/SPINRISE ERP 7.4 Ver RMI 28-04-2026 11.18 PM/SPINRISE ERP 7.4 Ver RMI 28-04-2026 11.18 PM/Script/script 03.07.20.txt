/*  
    SP_RM_Salesdaybook '01','2020','2020-06-01','2020-06-29' ,'C1A0001','D2Y0001','A'  
*/  
  
alter PROCEDURE SP_RM_Salesdaybook  
(  
@Divcode VARCHAR (02),  
@lotyear VARCHAR (10),  
@Fdate VARCHAR (15),  
@Tdate VARCHAR (15),  
@FCust VARCHAR (15),  
@TCust VARCHAR (15),  
@lottype VARCHAR (200)  
)  
as  
Begin  
  
Select delivery_dt,c.lotdt,sum(b.value) value,product_code,a.inv_type,a.inv_no,date,cust_code,  
slname,a.agent_code,d.state_code,brok_code,  
--(matl_value/(select sum(isnull(INV_KGS,0))  from rm_cinvdt aa  where  aa.inv_no=a.inv_no and aa.inv_type=aa.inv_type and a.divcode=aa.divcode ))*sum(ISNULL(INV_KGS,0)) 
sum(b.value)
 matl_value   
,(total_amt/(select sum(isnull(INV_KGS,0))  from rm_cinvdt aa  where  aa.inv_no=a.inv_no and aa.inv_type=a.inv_type and a.divcode=aa.divcode ))*sum(ISNULL(INV_KGS,0))

--sum(b.value)+sum(isnull(b.cgstamt,0))+sum(isnull(b.sgstamt,0))+sum(isnull(b.igstamt,0))
 total_amt  
  
  
,sum(isnull(b.cgstamt,0)) cgstamt,  
sum(isnull(b.sgstamt,0))sgstamt,sum(isnull(b.igstamt,0))igstamt,  
sum(case when c.bblflg = 'B' then packs  else 0 end) as bales,  
sum(case when c.bblflg   IN ('R','H') then packs else 0 end) as boras,rate_nett,  
sum(nett_kgs) nett_kgs,sum(ISNULL(INV_KGS,0)) AS INV_KGS, c.varcode,sum(isnull(b.cess_amt,0)) cess_amt,  
c.catcd,  
--vatamt=(case when d.state_code=s.state_code then round(isnull(b.tax_amt,0),0) else 0 end),s.state_code as divstate,  
--cstamt=(case when d.state_code<>s.state_code then round(isnull(b.tax_amt,0),0) else 0 end),  
r.VARNAME,s.div_printname,s.div_unitname,right(a.inv_no,3) as billno,  
(a.TCS_AMT/(select sum(isnull(INV_KGS,0))  from rm_cinvdt aa  where  aa.inv_no=a.inv_no and aa.inv_type=a.inv_type and a.divcode=aa.divcode ))*sum(ISNULL(INV_KGS,0))  TCS_AMT  
,(a.FREIGHT/(select sum(isnull(INV_KGS,0))  from rm_cinvdt aa  where  aa.inv_no=a.inv_no and aa.inv_type=a.inv_type and a.divcode=aa.divcode ))*sum(ISNULL(INV_KGS,0))  FREIGHT  
from rm_cinvhd a  
inner join rm_cinvdt b  on a.divcode=b.divcode and a.inv_type=b.inv_type and a.inv_no=b.inv_no  
inner join(select distinct lotno,lotdt,varcode,lottype,divcode,docno,docdt from rm_issb )bb on  b.product_code=bb.lotno  and b.lotdt=bb.lotdt and bb.divcode=a.divcode and bb.docno=a.bo_no and bb.docdt=a.bo_dt  
inner join rm_lot c on b.product_code=c.lotno  and a.divcode=c.divcode and b.LotDt=c.lotdt   
and c.lotno=bb.lotno and bb.lotdt=c.lotdt and bb.varcode=c.varcode and bb.lottype=c.lottype  
inner join fa_slmas d on  a.cust_code=d.slcode   
inner join pp_divmas s on a.divcode=s.divcode   
inner join rm_var r  on r.varcode = c.varcode   
where   
 a.divcode=@Divcode and c.lotyear= @lotyear AND    
Date Between @Fdate AND @Tdate and cust_code between @FCust and @TCust  AND a.inv_type in (select  items from dbo.split( @lottype ,',')) 
group by r.VARNAME,s.div_printname,s.div_unitname,a.inv_no ,c.catcd,c.bblflg,delivery_dt,c.lotdt,product_code,a.inv_no,date,cust_code  
,slname,a.agent_code,d.state_code,brok_code,matl_value,total_amt,rate_nett, c.varcode,a.inv_type,a.divcode,a.FREIGHT,a.TCS_AMT  
order by DATE  
  
End  
  
--select sum(CGSTAMT),sum(sGSTAMT),sum(iGSTAMT),sum(MATL_VALUE),sum(TOTAL_AMT) from rm_cinvhd   where date>='2020-06-01' and sGSTAMT<>CGSTAMT

--inner join rm_cinvdt b  on a.divcode=b.divcode and a.inv_type=b.inv_type and a.inv_no=b.inv_no

go

/*  
  MixingCost_abs '01','2020-06-01','2020-06-23','2020','po'                
*/  
CREATE PROC MixingCost_abs                           
(                    
 @Divcode varchar(5),                   
 @FromDate varchar(15),                   
 @ToDate varchar(15),             
 @LOTYEAR VARCHAR(10),    
 @IssueType varchar(300)               
)                
as                            
BEGIN                  
select distinct X.varcode,cntcd,x.cntname,docno,x.docdt,X.Catcd,X.lotno,X.lotdt,slname,  
count(Bales) AS BALE,UNIT,SUM(Kgs) AS netkg, isnull(ratekg,0) as ratekg,SUM(Value) AS Amount,  
AVG(ratecandy) AS RATECANDY,bblflg,x.DIVNAME,x.ADD1,x.DIV_PRINTNAME,x.DIV_UNITNAME,rv.VARNAME,  
sum(x.GRWT) as GRSWGT,sum(isnull(x.TAREWT,0)) TAREWT FROM(select distinct h.cntcd,b.Catcd,a.lotno,  
a.lotdt,b.supcd,unit=(case when bblflg='B' then 'Bales' else 'Borah' end),B.varcode,a.docno,  
(a.baleno) as Bales,bblflg,a.DOCDT,(A.ISSKGS) as Kgs,d.cntname,      
case when weightflg='M' then rmb.GRWT else rmb.PGRSWT end GRWT ,      
case when weightflg='M' then  rmb.TAREWT else rmb.PTAREWT end TAREWT,ratekg,   
ROUND((ratekg)*355.6187,0) as ratecandy,ROUND(ratekg*(a.ISSKGS),2) as VALUE,p.DIVNAME      
,p.ADD1,p.DIV_PRINTNAME,p.DIV_UNITNAME                  
From rm_issb a       
left join rm_lot b on  a.lotno=b.lotno and a.catcd=b.catcd and a.lotdt=b.lotdt and   
a.divcode=b.divcode and a.Varcode=b.VARCODE      
inner join rm_issh h on  a.docno=h.docno and a.docdt=h.docdt and a.isstype=h.isstype      
and a.divcode=h.divcode      
inner join RM_COUNT AS d on   h.divcode=d.divcode AND h.cntcd = d.cntcd         
inner join PP_DIVMAS p on   a.DIVCODE=p.DIVCODE        
left join rm_bale rmb on  a.lottype=rmb.lottype and a.lotno=rmb.lotno and a.lotdt=rmb.lotdt       
and a.catcd=rmb.catcd and a.baleno=rmb.baleno   and a.Varcode=rmb.VARCODE and rmb.DIVCODE=a.divcode      
Where B.LOTYEAR=@LOTYEAR   AND H.divcode=@Divcode             
AND H.DOCDT Between @FromDate and  @ToDate   
-- and h.ISSTYPE in (select items from dbo.split(@IssueType,','))                 
--group by h.cntcd,b.Catcd,a.lotno,a.lotdt,b.supcd,B.varcode,bblflg,ratekg,a.docno,a.docdt,p.DIVNAME,p.ADD1      
--,p.DIV_PRINTNAME,p.DIV_UNITNAME,d.cntname,rmb.GRWT,rmb.TAREWT       
)X      
left join   fa_slmas f on x.supcd=f.slcode       
left join RM_VAR rv    on   x.varcode=rv.VARCODE              
GROUP BY X.varcode,cntcd,X.Catcd,X.lotno,X.lotdt,slname,docno,x.docdt,      
bblflg,ratekg,UNIT,x.DIVNAME,x.ADD1,x.DIV_PRINTNAME,x.DIV_UNITNAME,rv.VARNAME,  
x.cntname--,x.GRWT,x.TAREWT                 
ORDER BY X.LOTNO                  
      
END
