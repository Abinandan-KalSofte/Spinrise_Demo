create View VW_IssueDetails  
as  
select distinct a.DIVCODE, a.docno,a.docdt,b.lotno,b.lotdt, lot.ARRNO ,lot.arrdt, b.varcode,lot.Line_No,lot.SUPCD ,  
sl.slname ,v.varname,bale.BALENO ,bale.PGRSWT PGRWT ,bale.GRWT,bale.GRWT-bale.PGRSWT Diffgrswt,  
bale.MOITURE inv_MOI,lot.PlCode,pl.PLName ,lot.billno,lot.billdt,bale.favaourablewgt NETWT ,bale.Comm_Wt   
from rm_issh a   
inner join rm_issb b on a.docno=b.docno and a.docdt=b.docdt and a.divcode=b.divcode  
inner join rm_lot lot on lot.lotno=b.lotno and lot.lotdt=b.lotdt and lot.divcode=b.divcode and b.varcode=lot.varcode and b.lottype=lot.lottype  
inner join RM_BALE bale on bale.BALENO =b.BALENO and b.LOTNO=bale.lotno and b.BALENO = bale.BALENO and b.lotdt =bale.lotdt  and b.VARCODE =bale.VARCODE   
and b.DIVCODE =bale.DIVCODE and b.lottype=bale.lottype  
inner join FA_SLMAS sl on sl.slcode=lot.SUPCD   
inner join RM_VAR v  on v.varcode=b.varcode  
left join RM_Plant pl on pl.PLCode =lot.PlCode  
left join (select IssDocno ,IssDocdt ,Divcode  from RM_FibreTestHD )ts on ts.divcode=a.divcode and ts.IssDocno=a.docno and ts.IssDocdt=a.DOCDT
where a.docdt>='2021-04-01'  and ts.IssDocno is null
  
 
go

create proc KSP_FibreTestResult
(
@Divcode Varchar(2),
@Fdate varchar(10),
@Tdate varchar(10),
@IPADD VArchar(18)
)
as
begin
if object_ID('RPT_Fibre_Test_Result') is not null drop table RPT_Fibre_Test_Result
select 
convert(varchar,a.IssDocdt ,3) Test_date,sl.slname 	Supplier,V.VARNAME 	Fibre,pl.PLName 	Plant,a.Line_no 	Line	, a.lotno Mill_lot_No,a.billno	Invoice_No,
convert(varchar,a.lotdt ,3) Date_of_Production,a.Batch_No 	Batch_No,BaleNO	Bale_No,	INVGrssWt Gross_Wt_Invoice 	,ActGrossWt  Gross_Wt_Actual ,GrossWtDiff 	Weight_Diff_Kgs,
InvoiceMoiture 'Invoice_Moisture %'	,MRHOTAIRAVG 'MR% (Hot Air)',MROHAUS 'MR% (OHAUS)',INVVsHOTAir	'Invoice Vs. Hot Air',InvVsOHAUS	'Invoice Vs. OHAUS', 
MRHOTAIR1 'MR% (Hot Air) 1',	MRHOTAIR2'MR% (Hot Air) 2'
,MROHAUS1 'MR% (OHAUS)1',MROHAUS2 'MR% (OHAUS)2',MROHAUS3 'MR% (OHAUS)3',MROHAUS4 'MR% (OHAUS)4',MROHAUS5 'MR% (OHAUS)5',MROHAUS6 'MR% (OHAUS)6',MROHAUS7 'MR% (OHAUS)7',MROHAUS8 'MR% (OHAUS)8',
MROHAUS9 'MR% (OHAUS)9', MROHAUS10 'MR% (OHAUS)10'

into RPT_Fibre_Test_Result
from RM_FibreTestHD a 
inner join RM_FibreTestDT b on a.Docno =b.Docno and a.Docdt =b.Docdt and a.Divcode =b.Divcode 
inner join FA_SLMAS sl on sl.slcode=a.slcode   
inner join RM_VAR v  on v.varcode=b.varcode  
left join RM_Plant pl on pl.PLCode =a.PlCode 
where a.Docdt between @Fdate and @Tdate and a.Divcode =@Divcode  
and a.varcode in (select varcode from RM_ProductprintDet where divcode =@Divcode and IPadd =@IPADD )
and a.slcode in (select supcd  from RM_ProductprintDet where divcode =@Divcode and IPadd =@IPADD )
end
