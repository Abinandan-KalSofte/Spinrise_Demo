BEGIN   

SELECT a.sno,A.CONTNO,A.CONTDT,A.SUPCD,A.ORDQTY,A.CANDYRATE,A.DLYTYPE,A.BBFLAG,a.MILLREFNO,                                                                                                                                                                                                                                      
case when (select max( DELDATE)from rm_delsched dd where dd.CONTNO=a.contno and dd.CONTDATE=a.contdt and dd.divcode=a.divcode) is null then                        
A.duedate else                        
(select max( DELDATE)from rm_delsched dd where dd.CONTNO=a.contno and dd.CONTDATE=a.contdt and dd.divcode=a.divcode )                        
end duedate,A.PAYTERMS,A.rateunit,A.remarks,A.contperson,rs.SEANAME season,A.FinalWeighment,A.TransPort,

(select max(name) from ig_state jj where b.StateCode=jj.state_code) 'B_STATE',B.COUNTRY B_COUNTRY,                                                            
f.TNGST, f.TNGSTDT, f.CGSTDT, f.CGST,C.slname 'D_Slname',C.add1 'D_Add1',C.add2 'D_Add2',                                                                                             
C.add3 'D_Add3',C.city 'D_City',C.pin 'D_Pin',(select max(name) from ig_state jj where c.StateCode=jj.state_code)                          
'D_STATE',C.Country 'D_COUNTRY',D.AREANAME,E.VARNAME,F.DIVCODE,F.PINCODE,f.HOState,F.PHONE1,F.FAX,                                                                                         
'cotton@jattechnospintexmill.com' AS EMAIL,F.WEBADDR,f.cinno,f.ecc,F.TINNO,f.DIV_PRINTNAME,F.DIV_UNITNAME,                                                          
--F.CGSTDT,F.CGST,A.MILLREFNO,J.[NAME],c.StateCode,K.NAME,F.HOADD2,F.HOEmail,H.REMARKS,a.contperson,

G.phone1 AS S_Phone1,g.TINNO AS s_tin,g.cgst AS s_cst,G_1.slname as A_slname,G_1.add1 AS A_ADD1,                               
G_1.add2 AS a_aDD2,G_1.add3 AS a_ADD3,G_1.city AS A_cITY,G_1.pin AS A_PIN,G_1.state AS A_STATE,                                                          
G_1.country AS A_COUNTRY,H.STAPLEN25,H.TRASH_PER,H.MICRONAIRE,H.MOITURE,H.GTEX,h.COLOURGRADE,

THEN (SELECT APP.[SIGNATURE] from  PO_ParaPOApproval APP WHERE a.Appuserid=APP.AppUserId AND app.AppLevel='5 - Final Level' AND app.divcode = @DIVCODE                                                                                                                                                                                            
) END AS FinalAppSign,a.appdate                                                                     
    
FROM  RM_CONT A                                                                                       
LEFT OUTER JOIN RM_BILLADD B ON  A.billadd = B.slcode                                                                                              
LEFT OUTER JOIN RM_deladd C ON  A.deladd = C.slcode -- and c.divcode=a.divcode
