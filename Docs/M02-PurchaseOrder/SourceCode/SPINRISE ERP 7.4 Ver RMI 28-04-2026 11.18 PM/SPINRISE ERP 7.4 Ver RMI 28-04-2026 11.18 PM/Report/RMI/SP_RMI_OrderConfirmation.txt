ALTER PROCEDURE SP_RMI_OrderConfirmation        
        
--EXEC SP_RMI_OrderConfirmation '01',59,'09-08-2009','S6'        
        
@Divcode VARCHAR(3)='',        
@ContNo VARCHAR(10)='',        
@ContDt VARCHAR(15),        
@VarCode VARCHAR(10)=''        
        
AS        
        
SELECT [C].CONTNO,[C].CONTDT,[C].VARCODE,[P].DIVCODE,[S].slcode AS SELLERCODE,[S].slname AS SELLERNAME,[S].add1 AS SELLERADD1,[S].add2 AS SELLERADD2,[S].add3 AS SELLERADD3,[S].CITY AS SELLERCITY,        
[S].pin  AS SELLERPINCODE,[S].state AS SELLERSTATE,[S].country AS SELLERCOUNTRY,[S].phone1 AS SELLERPHONE1,[S].phone2 AS SELLERPHONE2        
,[C].VARCODE AS DESCGOODS,[C].ORDQTY AS QUANTITY,[C].RATEkg AS PRICE,[C].DUEDATE AS DELVIPERIOD,(CASE WHEN [C].DLYTYPE='S' THEN 'SPOT' ELSE 'FOR' END) AS DELIVERYTYPE        
,[C].BRKCD AS BRKCODE        
,[B].SLNAME AS BRKNAME        
,[PY].paydesc AS PAYDESCRIPTION         
,(CASE WHEN [C].DLYTYPE='F' THEN 'SELLER A/C' ELSE 'MILL A/C' END) AS TRANSITINSURANCE        
,(CASE WHEN [C].DLYTYPE='F' THEN 'SELLER A/C' ELSE 'MILL A/C' END) AS LORRYFREIGHT        
,[C].acceptance AS COTTONAPPR        
,[AC].ANAME AS COTTONAPPRnAME        
,[C].CONTDT AS COTTONAPPRDT        
,[C].PAYMODE AS PAYMODE        
,[C].CREDITDAYS AS PAYDAYS        
, [P].HODIVNAME DIVNAME,[P].HOADD1 AS DIVADD1,[P].HOADD2 AS DIVADD2,[P].HOADD3 AS DIVADD3,[P].HOPIN AS DIVPIN        
,[P].HOCITY AS DIVCITY,[P].HOPHONE AS DIVPHONE        
FROM RM_CONT [C]          
INNER JOIN FA_SLMAS [S] ON [S].slcode=[C].SUPCD        
INNER JOIN RM_PMODE [PY] ON [PY].paymode=[C].PAYMODE        
LEFT OUTER JOIN FA_SLMAS [B] ON [B].slcode=[c].BRKCD        
INNER JOIN PP_DIVMAS [P] ON [P].DIVCODE=[C].DIVCODE AND [P].DIVCODE=[C].DIVCODE        
INNER JOIN RM_ACCEPTANCE [AC] ON [AC].DivCode=[C].DIVCODE AND [AC].CODE=[C].acceptance         
WHERE [P].DIVCODE=@Divcode  AND [C].CONTNO=@ContNo AND [C].CONTDT=@ContDt AND [C].VARCODE=@VarCode
