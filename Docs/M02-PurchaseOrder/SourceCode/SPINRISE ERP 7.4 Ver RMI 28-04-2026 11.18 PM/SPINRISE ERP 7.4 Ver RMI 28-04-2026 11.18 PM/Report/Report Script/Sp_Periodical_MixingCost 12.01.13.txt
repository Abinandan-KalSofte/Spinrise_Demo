  
--Sp_Periodical_MixingCost '01','2012-10-01','2012-10-30','2012'  
alter PROC Sp_Periodical_MixingCost          
(  
 @Divcode varchar(5),  
 @FromDate varchar(15),  
 @ToDate varchar(15),@LOTYEAR VARCHAR(10)  
)  
as          
BEGIN  
select distinct X.varcode,cntcd,docno,x.docdt,X.Catcd,X.lotno,X.lotdt,slname,SUM(Bales) AS BALE,UNIT,    
 SUM(Kgs) AS netkg, isnull(ratekg,0) as ratekg,SUM(Value) AS Amount,AVG(ratecandy) AS RATECANDY,    
 bblflg,x.DIVNAME,x.ADD1,x.DIV_PRINTNAME,x.DIV_UNITNAME,rv.VARNAME
 FROM(     
  select distinct cntcd,b.Catcd,a.lotno,a.lotdt,b.supcd,    
  Unit=(case when bblflg='B' then 'Bales' else 'Borah' end),B.varcode,a.docno,count(baleno) as Bales,bblflg,a.DOCDT,    
  SUM(A.ACTisskgs) as Kgs,     
  ratekg, ROUND(avg(ratekg)*355.616,0) as ratecandy,ROUND(ratekg*sum(a.ACTisskgs),2) as VALUE,p.DIVNAME,p.ADD1,p.DIV_PRINTNAME,p.DIV_UNITNAME    
 from rm_issb a,rm_lot b,rm_issh h,PP_DIVMAS p where B.LOTYEAR=@LOTYEAR    
  AND a.docno=h.docno and a.docdt=h.docdt and a.isstype=h.isstype and a.divcode=h.divcode  AND a.DIVCODE=p.DIVCODE  
 and a.lotno=b.lotno and a.catcd=b.catcd and a.lotdt=b.lotdt and a.divcode=b.divcode AND ISSUED='Y'     
 AND H.divcode=@Divcode AND H.DOCDT Between @FromDate and  @ToDate     
 group by cntcd,b.Catcd,a.lotno,a.lotdt,b.supcd,B.varcode,bblflg,ratekg,a.docno,a.docdt,p.DIVNAME,p.ADD1,p.DIV_PRINTNAME,p.DIV_UNITNAME  )X, fa_slmas f,RM_VAR rv
    
 where x.supcd=f.slcode   AND x.varcode=rv.VARCODE 
 GROUP BY X.varcode,cntcd,X.Catcd,X.lotno,X.lotdt,slname,docno,x.docdt,bblflg,ratekg,UNIT,x.DIVNAME,x.ADD1,x.DIV_PRINTNAME,x.DIV_UNITNAME,rv.VARNAME    
 ORDER BY x.docno,X.VARCODE,X.CNTCD,X.LOTNO    
END
