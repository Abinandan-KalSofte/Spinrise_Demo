ALTER PROC KSP_RMI_STOCKAGEING
@DIVCODE VARCHAR(5),
@TDATE VARCHAR(10)
AS
DECLARE @Mfdate  DATETIME ,@TLotYear VARCHAR(10),@FYFDATE DATETIME,@FYLDATE DATETIME
set @Mfdate = convert( varchar,YEAR(@TDATE))+'-'+convert(VARCHAR,MONTH(@TDATE))+'-01'
SELECT @FYFDATE=AYFDATE,@FYLDATE=AYLDATE,@TLotYear=YEAR(AYFDATE)  FROM PP_YEAR 
WHERE @TDATE BETWEEN AYFDATE AND AYLDATE

SELECT X.DIVCODE, ISNULL(X.VARCODE,'')VARCODE,SUM(X.RECBAL)-SUM(x.IssBAL)BALES,SUM(X.VAL)VAL,
x.LOTDT INTO #TMP1 FROM 
(
select B.DIVCODE,b.varcode,B.LOTNO,B.LOTDT,sum(isnull(b.bales,0)) AS Recbal,0 IssBal,SUM(b.favaourablewgt)*RATEKG AS VAL
 from rm_lot b where B.DIVCODE=@DIVCODE AND B.lotyear IN (@TLotYear) AND 
(b.lotdt <= @TDATE  or UPPER(OPFLG)='Y')and
 BALES >0   and (B.rejflg='N' or B.rejdt>=@FYFDATE) group by b.varcode,
 b.LOTNO , b.LOTDT, b.bblflg, b.ratekg,B.DIVCODE,b.RATEKG
 UNION ALL
 select B.DIVCODE,b.varcode,B.LOTNO,B.LOTDT,0 as RecBal,count(c.baleno) as issbal,(SUM(C.ISSKGS)*B.RATEKG)*-1 AS VAL
   from rm_lot b,rm_issb c 
 where C.ISSUED='Y' AND B.DIVCODE=C.DIVCODE AND B.DIVCODE=@DIVCODE AND 
 (b.lotdt <= @TDATE OR UPPER(OPFLG)='Y')  
 and b.lotno=c.lotno and B.lotyear in (@TLotYear) and b.lotdt=c.lotdt and b.lottype=c.lottype and b.catcd=c.catcd  
 and docdt BETWEEN @FYFDATE  AND @TDATE
 group by b.varcode,B.LOTNO,B.LOTDT,b.bblflg,b.ratekg,B.DIVCODE,B.RATEKG
 
UNION ALL
SELECT B.DIVCODE,E.VARCODE, e.LOTNO,e.lotdt,0 as RecBal,count(*) as issbal,(SUM(E.favaourablewgt)*B.RATEKG)*-1 AS VAL
from rm_bale e,
 rm_lot b where e.rejflg='Y' and e.rejdt <= @TDATE
 AND e.DIVCODE=@DIVCODE AND b.LOTYEAR=@TLotYear
 and e.lotno=b.lotno  and e.lotdt=b.lotdt and e.lottype=b.lottype and e.catcd=b.catcd 
 AND Status='RJ' group by e.varcode,e.LOTNO,e.lotdt,B.DIVCODE,B.RATEKG
)x GROUP BY x.varcode,x.LOTDT,X.DIVCODE
HAVING X.varcode<>'' AND SUM(X.RECBAL)-SUM(x.IssBAL)>0


SELECT X.VARCODE,RV.VARNAME,SUM(X.BALES1)BALES1,SUM(X.BALES2)BALES2,SUM(X.BALES3)BALES3,SUM(X.BALES4)BALES4,SUM(X.BALES5)BALES5,
SUM(X.VAL1)VAL1,SUM(X.VAL2)VAL2,SUM(X.VAL3)VAL3,SUM(X.VAL4)VAL4,SUM(X.VAL5)VAL5
INTO #TMP2 FROM
(
SELECT VARCODE,SUM(BALES)BALES1,0 BALES2,0 BALES3,0 BALES4,0 BALES5, 
SUM(VAL)VAL1,0 VAL2,0 VAL3,0 VAL4,0 VAL5 
FROM #TMP1  WHERE LOTDT 
BETWEEN DATEADD(M,-3,@TDATE) AND @TDATE AND DIVCODE =@DIVCODE GROUP BY VARCODE
UNION ALL
SELECT VARCODE,0 BALES1,SUM(BALES)BALES2,0 BALES3,0 BALES4,0 BALES5,
0 VAL1,SUM(VAL)VAL2,0 VAL3,0 VAL4,0 VAL5
 FROM #TMP1 WHERE LOTDT BETWEEN DATEADD(M,-6,@TDATE) AND DATEADD(M,-3,@TDATE) AND DIVCODE =@DIVCODE GROUP BY VARCODE
UNION ALL
SELECT VARCODE,0 BALES1,0 BALES2,SUM(BALES)BALES3,0 BALES4,0 BALES5,
0 VAL1,0 VAL2,SUM(VAL)VAL3,0 VAL4,0 VAL5
FROM #TMP1
WHERE LOTDT BETWEEN DATEADD(M,-9,@TDATE) AND DATEADD(M,-6,@TDATE) AND DIVCODE =@DIVCODE GROUP BY VARCODE
UNION ALL
SELECT VARCODE,0 BALES1,0 BALES2,0 BALES3,SUM(BALES)BALES4,0 BALES5,
0 VAL1,0 VAL2,0 VAL3,SUM(VAL)VAL4,0 VAL5
FROM #TMP1
WHERE LOTDT BETWEEN DATEADD(M,-12,@TDATE) AND DATEADD(M,-9,@TDATE) AND DIVCODE =@DIVCODE GROUP BY VARCODE
UNION ALL
SELECT VARCODE,0 BALES1,0 BALES2,0 BALES3,0 BALES4,SUM(BALES) BALES5,
0 VAL1,0 VAL2,0 VAL3,0 VAL4,SUM(VAL) VAL5
FROM #TMP1
WHERE LOTDT <=DATEADD(M,-12,@TDATE) AND DIVCODE =@DIVCODE GROUP BY VARCODE
)X,RM_VAR rv WHERE RV.VARCODE=X.VARCODE GROUP BY  X.VARCODE,RV.VARNAME
SELECT * FROM #TMP2
