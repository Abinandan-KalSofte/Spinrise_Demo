CREATE PROC SP_RMI_OrderConfirmation  
  
--EXEC SP_RMI_OrderConfirmation '01',59,'09-08-2009','S6'  
  
@Divcode VARCHAR(3)='',  
@ContNo VARCHAR(10)='',  
@ContDt VARCHAR(15)='',  
@VarCode VARCHAR(10)=''  
  
AS  
  
SELECT [C].CONTNO,[C].CONTDT,[A].VARCODE,[P].DIVCODE,[S].slcode AS SELLERCODE,[S].slname AS SELLERNAME,[S].add1 AS SELLERADD1,[S].add2 AS SELLERADD2,[S].add3 AS SELLERADD3,[S].CITY AS SELLERCITY,  
[S].pin  AS SELLERPINCODE,[S].state AS SELLERSTATE,[S].country AS SELLERCOUNTRY,[S].phone1 AS SELLERPHONE1,[S].phone2 AS SELLERPHONE2  
,[A].VARCODE AS DESCGOODS,[A].QUANTITY AS QUANTITY,[A].RATECY AS PRICE,[C].DUEDATE AS DELVIPERIOD,(CASE WHEN [C].DLYTYPE='S' THEN 'SPOT' ELSE 'FOR' END) AS DELIVERYTYPE  
,[A].BRKCD AS BRKCODE  
,[B].SLNAME AS BRKNAME  
,[PY].paydesc AS PAYDESCRIPTION   
,(CASE WHEN [C].DLYTYPE='S' THEN 'SELLER A/C' ELSE 'MILL A/C' END) AS TRANSITINSURANCE  
,(CASE WHEN [C].DLYTYPE='S' THEN 'SELLER A/C' ELSE 'MILL A/C' END) AS LORRYFREIGHT  
,[C].acceptance AS COTTONAPPR  
,[AC].ANAME AS COTTONAPPRnAME  
,[C].CONTDT AS COTTONAPPRDT  
,[C].PAYMODE AS PAYMODE  
,[C].CREDITDAYS AS PAYDAYS  
, [P].DIVNAME,[P].ADD1 AS DIVADD1,[P].ADD2 AS DIVADD2,[P].ADD3 AS DIVADD3,[P].PINCODE AS DIVPIN  
,[P].CITY AS DIVCITY,[P].PHONE1 AS DIVPHONE  
FROM RM_ARRIVAL [A]    
INNER JOIN RM_CONT  [C] ON [A].DIVCODE=[C].DIVCODE AND [A].CONTNO=[C].CONTNO AND [A].ARRDATE=[C].CONTDT  
INNER JOIN FA_SLMAS [S] ON [S].slcode=[C].SUPCD  
INNER JOIN RM_PMODE [PY] ON [PY].paymode=[C].PAYMODE  
LEFT OUTER JOIN FA_SLMAS [B] ON [B].slcode=[A].BRKCD  
INNER JOIN PP_DIVMAS [P] ON [P].DIVCODE=[C].DIVCODE AND [P].DIVCODE=[C].DIVCODE  
INNER JOIN RM_ACCEPTANCE [AC] ON [AC].DivCode=[C].DIVCODE AND [AC].CODE=[C].acceptance   
WHERE [P].DIVCODE=@Divcode  AND [C].CONTNO=@ContNo AND [C].CONTDT=CONVERT(DATETIME,@ContDt,103) AND [A].VARCODE=@VarCode
