(    
@Divcode VArchar(25),    
@Fdate Varchar(15),    
@Tdate Varchar(15),    
@IPADD Varchar(15)    
)    
as

a.ContNum,a.BILLNO ,    
case when cn.im_ind='I' then 'Import'     
when cn.im_ind='L' then 'Domestic'    
when cn.im_ind='U' then 'Interstate' End InvoiceType,    
a.BILLDT ,a.Comm_Wt ,CurrCode,e.PLName 'Plant',a.Line_No,'' Color ,a.Bill_Rate ,    
--BDType1,BDType2,BDType3,BDType4,BDType5,BDType6,BDType7,BDType8,

,a.RATECY,

DRate1,DRate2,DRate3,DRate4,DRate5,DRate6,DRate7,DRate8    
,a.Bill_Rate - a.RATECY Elig_Diff_Inv_Rate,
