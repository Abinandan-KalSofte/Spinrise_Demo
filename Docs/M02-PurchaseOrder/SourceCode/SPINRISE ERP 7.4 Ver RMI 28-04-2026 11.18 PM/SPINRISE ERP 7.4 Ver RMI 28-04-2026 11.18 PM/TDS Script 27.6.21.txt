create proc KSP_TDSAmountCalculation
(
@Divcode varchar(2),
@Yfdate Varchar(15),
@Yldate Varchar(15),
@Slcode varchar(8),
@Pdate varchar(18)

)
as
select sum(Billamt) Billamt from fa_purhd  a
inner join fa_slmas b on a.slcode=b.slcode
where a.pjvdt between @Yfdate and @Yldate and b.slcode=@Slcode-- b.pan=(select pan from fa_slmas where slcode=@Slcode)
and a.pjvdt<=@Pdate
go
