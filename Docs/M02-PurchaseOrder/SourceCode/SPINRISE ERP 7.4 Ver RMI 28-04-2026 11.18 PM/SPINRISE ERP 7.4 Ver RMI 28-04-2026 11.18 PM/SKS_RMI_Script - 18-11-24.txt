alter proc Ksp_RMI_supplier_list                    
(                    
@divcode Varchar(5),                    
@ptype Varchar(5)                    
)                    
AS                    
BEGIN                     
SELECT a.SLCODE,a.SLNAME,a.phone1 as phone,a.EMAIL,a.add1,a.add2,a.add3,a.city,a.pin,state,            
a.country,b.divname as divname,b.div_unitname,a.gstinno as tngst,a.cgst,a.ecc,a.gstinno,a.gststatecode            
FROM FA_SLMAS a               
inner join pp_divmas b on b.divcode=@divcode           
WHERE a.prtytyp='C1'  AND @ptype='C1'         
order by a.slcode                   
end   
  

go

alter PROC rm_sam          
(          
 @Divcode VARCHAR(3),            
 @SamNo NUMERIC(10),            
 @FDate VARCHAR(15)            
 )          
AS            
BEGIN            
 SELECT a.DIVCODE,a.SNO,a.DATE,a.SUPCODE, b.slname,b.add1 as sadd1,b.add2 as sadd2,b.add3 as sadd3,b.city as scity,
 b.pin as spin,b.[state] as sstate,b.country as scountry,b.phone1 as sphone,b.gstinno as sgstinno,b.gststatecode as sgststcode,
 b.email as semail,b.phone2,a.VARCODE,c.varname,a.STAPLEN25,a.immature,a.yellowtouch,            
 a.STAPLEN50,a.TRASH_PER, a.MICRONAIRE,a.STRENGTH,a.MCOEFF,remarks as remarks1,a.avgcount, a.UNIRATIO,a.yarnapp,a.STAPHANDTEST,            
 a.HVITEST,a.moiture,a.tpi,a.csp ,a.tpi ,a.avrstrength , a.corr_strenth , a.spot_for , a.rate ,a.prno ,a.prty_sampno,            
 a.brokercode , a.station_code,POTYPE,GRADE,GTEX,COLOR,D.AREANAME,a.STATE_CODE,STATION_NAME,CROP_YEAR,              
 PD.DIVCODE,PD.DIV_PRINTNAME,PD.DIV_UNITNAME,PD.div_logo,a.REMARKS,pd.ADD1,pd.ADD2,pd.ADD3,pd.CITY,pd.PINCODE,pd.HOState,pd.HOCountry,pd.PHONE1,pd.PHONE2,pd.cust_id,          
 pd.PHONE3,pd.FAX,pd.EMAIL,pd.WEBADDR,    
 a.SHORTFIBRE,a.QUANTITY,a.plotno,br.slname AS agent,pd.gstinno,pd.CINNO    
 FROM RM_SAMPLE a          
 LEFT JOIN fa_slmas b ON a.supcode=b.slcode    
 LEFT JOIN fa_slmas br ON a.brokercode = br.slcode          
 INNER JOIN rm_var c ON a.varcode=c.varcode           
 INNER JOIN  RM_AREA d ON a.area_code = d.areacode          
 --INNER JOIN  fa_slmas e ON           
 inner join PP_DIVMAS pd ON  PD.DIVCODE =A.DIVCODE            
 WHERE              
 a.date = @FDate and a.sno=@SamNo and a.divcode=@Divcode             
END     


GO


  
alter Procedure Ksp_Materialtransferreceipt        
---Ksp_Materialtransferreceipt '01','109','2015-05-31','2015'        
@divcode varchar(02),        
@lotno varchar(20),        
@lotdate varchar(20),        
@lotyear varchar(15)        
as        
begin        
SELECT        
    RM_LOT."LOTNO",        
    RM_LOT."LOTDT",        
    RM_LOT."LOTTYPE",        
    RM_LOT."NETWT",        
    RM_LOT."BALES",        
    RM_LOT."PLOTNO",        
    RM_LOT."PNETWT",      
    pp_divmas."ADD1",        
    PP_DIVMAS."ADD2",        
    PP_DIVMAS."ADD3",        
    PP_DIVMAS."CITY",        
    PP_DIVMAS."PINCODE",      
    PP_DIVMAS."PHONE1",        
    PP_DIVMAS."FAX",        
    PP_DIVMAS."EMAIL",        
    PP_DIVMAS."WEBADDR",        
    PP_DIVMAS."HOState",        
    PP_DIVMAS."HOCountry",        
    PP_DIVMAS."DIV_PRINTNAME",  
	PP_DIVMAS."DIV_UNITNAME", 
    PP_DIVMAS."DIV_LOGO",        
    FA_SLMAS."slname",        
    FA_SLMAS."add1",        
    FA_SLMAS."add2",        
    FA_SLMAS."add3",        
    FA_SLMAS."city",        
    FA_SLMAS."pin",        
    FA_SLMAS."state",        
    FA_SLMAS."country",        
    rm_var."VARNAME",fa.slname agent,ra.AREANAME,rm_lot.brkcd,rm_lot.ARRDT,rm_lot.ARRNO ,
	PP_DIVMAS.gstinno, PP_DIVMAS.CINNO
FROM        
    RM_LOT RM_LOT         
    INNER JOIN PP_DIVMAS PP_DIVMAS ON        
        RM_LOT."DIVCODE" = PP_DIVMAS."DIVCODE"        
    left outer JOIN FA_SLMAS FA_SLMAS ON        
        RM_LOT."SUPCD" = FA_SLMAS."slcode"        
    left outer join rm_var rm_var ON        
        RM_LOT."VARCODE" = rm_var."VARCODE" AND        
        RM_LOT."CATCD" = rm_var."CATCD"        
        left join fa_slmas fa on fa.slcode=rm_lot.brkcd        
        left outer join rm_area ra on ra.AREACODE=rm_lot.AREACD        
where rm_lot.divcode=@divcode and rm_lot.LOTNO=@lotno and rm_lot.LOTDT=@lotdate  and rm_lot.LOTYEAR=@lotyear               
        
            
end        


go
